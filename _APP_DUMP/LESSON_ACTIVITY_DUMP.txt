<script>
(async function(){
  const levels=["Absolute Beginner","A0","A1","A1.1","A2","B1","B2"];
  const grid=document.getElementById("levelGrid");
  const createBtn=document.getElementById("createBtn");

  const templateBase=[
    {key:"speaking-starter",title:"Template 1 — Speaking Starter",objective:"Learners practise simple oral production from the first minutes of class.",flow:"Warm-up → teacher model → repetition → guided speaking → independent speaking → quick correction.",materials:"Speaking prompts, name cards, simple question frames."},
    {key:"vocabulary-builder",title:"Template 2 — Vocabulary Builder",objective:"Learners use new vocabulary confidently in short spoken sentences.",flow:"Picture prompt → vocabulary drilling → sentence building → oral practice → mini performance check.",materials:"Picture cards, vocabulary list, sentence frames."},
    {key:"assessment-prep",title:"Template 3 — Assessment Prep",objective:"Learners review weak areas and practise assessment-style tasks.",flow:"Quick diagnostic → correction → guided practice → individual attempt → feedback.",materials:"Review worksheet, assessment prompts, correction notes."}
  ];

  function impactConfirm(message){
    return new Promise(resolve=>{
      const modal=document.getElementById("impactConfirmModal");
      const msg=document.getElementById("impactConfirmMessage");
      const ok=document.getElementById("impactConfirmOk");
      const cancel=document.getElementById("impactConfirmCancel");

      msg.textContent=message;
      modal.classList.add("show");

      const close=(result)=>{
        modal.classList.remove("show");
        ok.onclick=null;
        cancel.onclick=null;
        modal.onclick=null;
        resolve(result);
      };

      ok.onclick=()=>close(true);
      cancel.onclick=()=>close(false);
      modal.onclick=(e)=>{ if(e.target===modal) close(false); };
    });
  }

  function deletedTemplates(){
    try{return JSON.parse(localStorage.getItem("impactDeletedLessonTemplates")||"[]")}
    catch(e){return []}
  }

  function saveDeletedTemplate(id){
    const deleted=deletedTemplates();
    deleted.push(id);
    localStorage.setItem("impactDeletedLessonTemplates",JSON.stringify([...new Set(deleted)]));
  }

  function templatePlansFor(level){
    const deleted=deletedTemplates();
    return templateBase.map(t=>({
      id:`template-${level.replace(/\s+/g,"-").replace(/\./g,"-").toLowerCase()}-${t.key}`,
      level,
      title:t.title,
      type:"template",
      isTemplate:true,
      objective:t.objective,
      flow:t.flow,
      materials:t.materials
    })).filter(t=>!deleted.includes(t.id));
  }

  async function safeGetPlans(){
    try{return await window.ImpactFirebaseStore.getLessonPlans()}
    catch(e){console.warn(e);return []}
  }

  async function render(){
    const savedPlans=await safeGetPlans();

    grid.innerHTML=levels.map(level=>{
      const files=[...templatePlansFor(level),...savedPlans.filter(p=>p.level===level)];
      const fileHtml=files.map(file=>`
        <div class="lesson-file-row">
          <button class="file-pill" data-id="${file.id}" data-template="${file.isTemplate ? "yes" : "no"}" type="button">${file.title || file.name || "Untitled Lesson"}</button>
          <button class="lesson-action-btn lesson-view-btn" data-id="${file.id}" data-template="${file.isTemplate ? "yes" : "no"}" type="button">🔎</button>
          <button class="lesson-action-btn lesson-edit-btn" data-id="${file.id}" data-template="${file.isTemplate ? "yes" : "no"}" type="button">✍🏽</button>
          <button class="lesson-delete-btn" data-id="${file.id}" data-template="${file.isTemplate ? "yes" : "no"}" type="button">❌</button>
        </div>`).join("");

      return `<article class="level-card" data-level="${level}">
        <h2>${level}</h2>
        <p>${files.length} lesson plan${files.length===1?"":"s"}</p>
        <div class="file-list">${fileHtml}</div>
      </article>`;
    }).join("");
  }

  grid.addEventListener("click",async e=>{
    const deleteBtn=e.target.closest(".lesson-delete-btn");
    const viewBtn=e.target.closest(".lesson-view-btn,.file-pill");
    const editBtn=e.target.closest(".lesson-edit-btn");
    const card=e.target.closest(".level-card");

    if(deleteBtn){
      e.preventDefault();
      e.stopPropagation();

      const id=deleteBtn.dataset.id;
      const isTemplate=deleteBtn.dataset.template==="yes";

      if(isTemplate){
        const ok=await impactConfirm("Delete this template from the folder?");
        if(!ok) return;
        saveDeletedTemplate(id);
        await render();
        return;
      }

      const ok=await impactConfirm("Delete this lesson plan?");
      if(!ok) return;
      await window.ImpactFirebaseStore.deleteLessonPlan(id);
      await render();
      return;
    }

    if(viewBtn){
      e.preventDefault();
      e.stopPropagation();
      const row=viewBtn.closest(".lesson-file-row");
      const id=viewBtn.dataset.id;
      const level=viewBtn.closest(".level-card").dataset.level;
      const template=templatePlansFor(level).find(t=>t.id===id);
      const selected=template || {id,title:row.querySelector(".file-pill").textContent.trim(),level,type:"saved"};
      localStorage.setItem("impactSelectedLessonPlan",JSON.stringify(selected));
      sessionStorage.setItem("impactEditingLessonId",id);
      location.href="/teacher/lesson-plan-view.html?id="+encodeURIComponent(id);
      return;
    }

    if(editBtn){
      e.preventDefault();
      e.stopPropagation();
      const row=editBtn.closest(".lesson-file-row");
      const id=editBtn.dataset.id;
      const level=editBtn.closest(".level-card").dataset.level;
      const template=templatePlansFor(level).find(t=>t.id===id);
      const selected=template || {id,title:row.querySelector(".file-pill").textContent.trim(),level,type:"saved"};
      localStorage.setItem("impactSelectedLessonPlan",JSON.stringify(selected));
      sessionStorage.setItem("impactEditingLessonId",id);
      location.href="/teacher/lesson-builder.html?edit="+encodeURIComponent(id);
      return;
    }

    if(card){
      document.querySelectorAll(".level-card").forEach(c=>{if(c!==card)c.classList.remove("open")});
      card.classList.toggle("open");
      grid.classList.toggle("folder-focus-active",!!document.querySelector(".level-card.open"));
      sessionStorage.setItem("impactSelectedLessonLevel",card.dataset.level);
    }
  });

  document.addEventListener("click",e=>{
    if(e.target.closest(".level-card") || e.target.closest(".create-btn") || e.target.closest(".impact-confirm-card")) return;
    document.querySelectorAll(".level-card.open").forEach(card=>card.classList.remove("open"));
    grid.classList.remove("folder-focus-active");
  });

  createBtn.onclick=()=>{
    const selected=sessionStorage.getItem("impactSelectedLessonLevel") || "Absolute Beginner";
    location.href="/teacher/lesson-builder.html?level="+encodeURIComponent(selected);
  };

  await render();
})();
</script>

(function(){

  function injectTeamMeetings(){

    const menu =
      document.querySelector(".quick-menu") ||
      document.querySelector(".menu-dropdown") ||
      document.querySelector(".menu-panel") ||
      document.querySelector(".menu-links") ||
      document.querySelector(".dropdown-menu");

    if(!menu) return;

    if(document.getElementById("teamMeetingsMenuLink")) return;

    const recycleLink = Array.from(menu.querySelectorAll("a"))
      .find(a =>
        (a.textContent || "")
          .toLowerCase()
          .includes("recycle")
      );

    if(!recycleLink) return;

    const newLink = recycleLink.cloneNode(true);

    newLink.id = "teamMeetingsMenuLink";

    newLink.href = "/teacher/team-meetings.html";

    newLink.textContent = "Team Meetings";

    // REMOVE INLINE STYLES THAT CAUSED STATIC LOOK
    newLink.removeAttribute("style");

    // insert before recycle bin
    recycleLink.parentNode.insertBefore(newLink, recycleLink);

  }

  injectTeamMeetings();

  setInterval(injectTeamMeetings, 1500);

})();
</script>


<script id="lesson-plan-delete-fix">
document.addEventListener("DOMContentLoaded", function(){

  function getDb(){
    if(window.firebase && firebase.firestore){
      return firebase.firestore();
    }
    return null;
  }

  document.addEventListener("click", async function(e){

    const btn = e.target.closest(
      "[data-delete-id], [data-delete], .delete-plan, .delete-btn, .plan-delete-btn, button"
    );

    if(!btn) return;

    const label = (btn.textContent || "").trim().toLowerCase();
    const looksLikeDelete =
      label.includes("delete") ||
      label === "×" ||
      label === "x" ||
      btn.classList.contains("delete-plan") ||
      btn.classList.contains("delete-btn") ||
      btn.classList.contains("plan-delete-btn") ||
      btn.hasAttribute("data-delete") ||
      btn.hasAttribute("data-delete-id");

    if(!looksLikeDelete) return;

    e.preventDefault();
    e.stopPropagation();

    const row =
      btn.closest("[data-id]") ||
      btn.closest("[data-plan-id]") ||
      btn.closest(".lesson-plan-row") ||
      btn.closest(".lesson-plan-item") ||
      btn.closest(".plan-row") ||
      btn.closest(".saved-plan") ||
      btn.closest(".folder-plan") ||
      btn.closest("article") ||
      btn.closest("li") ||
      btn.closest(".card");

    const id =
      btn.dataset.deleteId ||
      btn.dataset.delete ||
      btn.dataset.id ||
      row?.dataset?.id ||
      row?.dataset?.planId;

    if(!id){
      alert("Could not find this lesson plan ID.");
      return;
    }

    if(!confirm("Delete this lesson plan?")) return;

    try{
      const db = getDb();

      if(db){
        const collections = [
          "lessonPlans",
          "lessons",
          "savedLessonPlans"
        ];

        for(const name of collections){
          try{
            await db.collection(name).doc(id).delete();
          }catch(err){}
        }
      }

      // Also remove from localStorage backup if the page uses local saved plans
      const keys = [
        "impactLessonPlans",
        "impactSavedLessonPlans",
        "lessonPlans"
      ];

      keys.forEach(key => {
        try{
          const list = JSON.parse(localStorage.getItem(key) || "[]");
          if(Array.isArray(list)){
            const next = list.filter(item => item.id !== id && item.planId !== id);
            localStorage.setItem(key, JSON.stringify(next));
          }
        }catch(err){}
      });

      if(row){
        row.remove();
      }else{
        window.location.reload();
      }

    }catch(err){
      console.error(err);
      alert("Delete failed. Check console for details.");
    }

  }, true);

});
</script>

// ── CREATE BLOCK ──────────────────────────────────────────
function createBlock(type, data){
  const tmpl = TEMPLATES[type];
  if(!tmpl) return null;

  const block = document.createElement("div");
  block.className = "c-block";
  block.setAttribute("data-type", type);
  block.setAttribute("draggable", "true");

  const actions = document.createElement("div");
  actions.className = "block-actions";
  actions.innerHTML = `
    <button class="act-btn" data-act="up" title="Move up">↑</button>
    <button class="act-btn" data-act="down" title="Move down">↓</button>
    <div class="act-divider"></div>
    <button class="act-btn" data-act="dupe" title="Duplicate">⧉</button>
    <div class="act-divider"></div>
    <button class="act-btn danger" data-act="del" title="Delete">✕</button>
  `;

  const inner = document.createElement("div");
  inner.className = "c-block-inner";
  inner.innerHTML = tmpl.render(data);

  block.appendChild(actions);
  block.appendChild(inner);

  bindBlock(block);
  return block;
}

function bindBlock(block){
  block.addEventListener("click", function(e){
    if(e.target.closest(".block-actions")) return;
    e.stopPropagation();
    selectBlock(block);
  });

  const actionBar = block.querySelector(".block-actions");
  actionBar.addEventListener("click", function(e){
    e.stopPropagation();
    const btn = e.target.closest("[data-act]");
    if(!btn) return;
    const act = btn.dataset.act;

    if(act==="del"){
      pushUndo();
      block.remove();
      selectedBlock = null;
      showEmpty();
    }
    if(act==="up"){
      pushUndo();
      const prev = block.previousElementSibling;
      if(prev && prev.classList.contains("c-block")) canvas.insertBefore(block, prev);
    }
    if(act==="down"){
      pushUndo();
      const next = block.nextElementSibling;
      if(next && next.classList.contains("c-block")) canvas.insertBefore(next, block);
    }
    if(act==="dupe"){
      pushUndo();
      const type = block.dataset.type;
      const dupe = createBlock(type, extractData(block));
      block.after(dupe);
      selectBlock(dupe);
    }
  });

  // Add option button for multiple choice
  const mcAdd = block.querySelector(".mc-add");
  if(mcAdd){
    mcAdd.addEventListener("click", function(e){
      e.stopPropagation();
      pushUndo();
      const grid = block.querySelector(".mc-options");
      const opt = document.createElement("div");
      opt.className = "mc-option";
      opt.innerHTML = `<input type="radio" disabled><span class="mc-option-label" contenteditable="true">Option ${grid.children.length+1}</span>`;
      grid.appendChild(opt);
    });
  }

  // Add image for picture choice
  const pcAdd = block.querySelector(".pc-add-img");
  if(pcAdd){
    pcAdd.addEventListener("click", function(e){
      e.stopPropagation();
      pcImageInput.click();
      pcImageInput.onchange = function(){
        const files = Array.from(pcImageInput.files);
        const grid = block.querySelector(".pc-grid");
        files.forEach(file=>{
          const reader = new FileReader();
          reader.onload = function(ev){
            const item = document.createElement("div");
            item.className = "pc-item";
            item.innerHTML = `<div class="pc-img-placeholder"><img src="${ev.target.result}" alt=""></div><div class="pc-item-label"><input type="radio" disabled><span contenteditable="true">Option</span></div>`;
            grid.appendChild(item);
          };
          reader.readAsDataURL(file);
        });
        pcImageInput.value="";
      };
    });
  }

  // Drag and drop reorder
  block.addEventListener("dragstart", function(e){
    dragSrc = block;
    e.dataTransfer.effectAllowed = "move";
    setTimeout(()=>block.style.opacity=".4", 0);
  });
  block.addEventListener("dragend", function(){
    block.style.opacity="";
    allBlocks().forEach(b=>b.classList.remove("drag-over"));
    dragSrc = null;
  });
  block.addEventListener("dragover", function(e){
    e.preventDefault();
    if(dragSrc && dragSrc !== block) block.classList.add("drag-over");
  });
  block.addEventListener("dragleave", function(){
    block.classList.remove("drag-over");
  });
  block.addEventListener("drop", function(e){
    e.preventDefault();
    block.classList.remove("drag-over");
    if(!dragSrc || dragSrc===block) return;
    pushUndo();
    const rect = block.getBoundingClientRect();
    if(e.clientY < rect.top + rect.height/2) canvas.insertBefore(dragSrc, block);
    else block.after(dragSrc);
  });
}

// ── SERIALISE / DESERIALISE ───────────────────────────────
function extractData(block){
  const type = block.dataset.type;
  const data = { type };

  if(type==="header"||type==="subheader"||type==="text"){
    data.content = block.querySelector("[contenteditable]").innerHTML;
  }
  if(type==="pagebreak"){
    const wrap = block.querySelector(".page-break-wrap");
    data.pageNum = wrap ? parseInt(wrap.dataset.page)||2 : 2;
  }
  if(type==="shorttext"||type==="longtext"||type==="date"||type==="number"||type==="website"||type==="fileupload"||type==="signature"||type==="mediaanswer"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
  }
  if(type==="multiplechoice"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.options = Array.from(block.querySelectorAll(".mc-option-label")).map(el=>el.innerHTML);
  }
  if(type==="picturechoice"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.images = Array.from(block.querySelectorAll(".pc-img-placeholder img")).map(img=>img.src);
  }
  if(type==="dropdown"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.options = Array.from(block.querySelectorAll("select option")).slice(1).map(o=>o.textContent);
  }
  return data;
}

function serializeBlocks(){
  return allBlocks().map(extractData);
}

function rebuildFromBlocks(blocks){
  allBlocks().forEach(b=>b.remove());
  selectedBlock = null;
  pageCount = 1;
  if(!blocks || !blocks.length){ showEmpty(); return; }
  hideEmpty();
  blocks.forEach(data=>{
    const block = createBlock(data.type, data);
    if(!block) return;
    canvas.appendChild(block);
    // Restore editable content
    if(data.content){
      const ce = block.querySelector("[contenteditable]");
      if(ce) ce.innerHTML = data.content;
    }
    if(data.label){
      const lel = block.querySelector(".q-label-text");
      if(lel) lel.innerHTML = data.label;
    }
    if(data.hint){
      const hel = block.querySelector(".q-hint");
      if(hel) hel.innerHTML = data.hint;
    }
  });
}

async function save(){
  const blocks = serializeBlocks();
  const payload = {
    title: titleInput.value.trim() || "Untitled Lesson",
    level: levelSelect.value,
    blocks: blocks,
    version: 2,
    updatedAt: new Date().toISOString()
  };
  if(currentId){
    await db.collection("lessonPlans").doc(currentId).set(payload, {merge:true});
  } else {
    const ref = await db.collection("lessonPlans").add(payload);
    currentId = ref.id;
    history.replaceState(null,"","/teacher/lesson-builder.html?edit="+currentId);
  }
  tick();
  return currentId;
}

<script>
document.addEventListener("DOMContentLoaded", async function(){
  const db = firebase.firestore();
  const wrap = document.getElementById("foldersWrap");
  const folderCount = document.getElementById("folderCount");
  let allPlans = [];
  let openFolder = null;

  function confirm(msg){
    return new Promise(resolve=>{
      const modal = document.getElementById("confirmModal");
      document.getElementById("confirmMsg").textContent = msg;
      modal.classList.add("show");
      const close = r=>{ modal.classList.remove("show"); resolve(r); };
      document.getElementById("confirmOk").onclick = ()=>close(true);
      document.getElementById("confirmCancel").onclick = ()=>close(false);
      modal.onclick = e=>{ if(e.target===modal) close(false); };
    });
  }

  async function loadPlans(){
    try{
      const snap = await db.collection("activityPlans").orderBy("updatedAt","desc").get();
      allPlans = snap.docs.map(d=>({id:d.id,...d.data()}));
    }catch(e){
      const snap = await db.collection("activityPlans").get();
      allPlans = snap.docs.map(d=>({id:d.id,...d.data()}));
    }
    render();
  }

  function render(){
    const folderMap = {};
    allPlans.forEach(plan=>{
      const lvl = (plan.level || "Uncategorised").trim();
      if(!folderMap[lvl]) folderMap[lvl] = [];
      folderMap[lvl].push(plan);
    });
    const levels = Object.keys(folderMap).sort();
    folderCount.textContent = levels.length ? `${levels.length} folder${levels.length===1?"":"s"}` : "";

    if(!levels.length){
      wrap.innerHTML = `<div class="empty-state"><h3>No folders yet</h3><p>Create an activity and select a level in the builder.<br>The folder will appear here automatically.</p></div>`;
      return;
    }

    wrap.innerHTML = `<div class="folders-grid" id="foldersGrid"></div>`;
    const grid = document.getElementById("foldersGrid");

    levels.forEach(level=>{
      const plans = folderMap[level];
      const card = document.createElement("div");
      card.className = "folder-card" + (openFolder===level?" open":"");
      card.dataset.level = level;
      const planRows = plans.map(plan=>`
        <div class="plan-row">
          <span class="plan-title">${plan.title||"Untitled Activity"}</span>
          <button class="plan-btn view-btn" data-id="${plan.id}" title="Preview">👁</button>
          <button class="plan-btn edit-btn" data-id="${plan.id}" title="Edit">✏️</button>
          <button class="plan-btn del" data-id="${plan.id}" data-title="${(plan.title||"Untitled Activity").replace(/"/g,"&quot;")}" title="Delete">✕</button>
        </div>`).join("");
      card.innerHTML = `
        <div class="folder-head">
          <span class="folder-icon">${openFolder===level?"📂":"📁"}</span>
          <div class="folder-meta">
            <div class="folder-name">${level}</div>
            <div class="folder-sub">${plans.length} activit${plans.length===1?"y":"ies"}</div>
          </div>
          <span class="folder-chevron">›</span>
        </div>
        <div class="folder-body">${planRows}</div>`;
      grid.appendChild(card);
    });
    bindEvents();
  }

  function bindEvents(){
    const grid = document.getElementById("foldersGrid");
    if(!grid) return;
    grid.addEventListener("click", async function(e){
      const delBtn = e.target.closest(".del");
      if(delBtn){
        e.stopPropagation();
        const ok = await confirm(`Delete "${delBtn.dataset.title||"this activity"}"?`);
        if(!ok) return;
        await db.collection("activityPlans").doc(delBtn.dataset.id).delete();
        allPlans = allPlans.filter(p=>p.id!==delBtn.dataset.id);
        const level = delBtn.closest(".folder-card").dataset.level;
        if(!allPlans.filter(p=>(p.level||"Uncategorised")===level).length) openFolder=null;
        render(); return;
      }
      const viewBtn = e.target.closest(".view-btn");
      if(viewBtn){ e.stopPropagation(); location.href="/teacher/activity-view.html?id="+encodeURIComponent(viewBtn.dataset.id); return; }
      const editBtn = e.target.closest(".edit-btn");
      if(editBtn){ e.stopPropagation(); location.href="/teacher/activity-builder.html?edit="+encodeURIComponent(editBtn.dataset.id); return; }
      const card = e.target.closest(".folder-card");
      if(card){
        const level = card.dataset.level;
        if(openFolder===level){ openFolder=null; card.classList.remove("open"); card.querySelector(".folder-icon").textContent="📁"; }
        else {
          document.querySelectorAll(".folder-card.open").forEach(c=>{ c.classList.remove("open"); c.querySelector(".folder-icon").textContent="📁"; });
          openFolder=level; card.classList.add("open"); card.querySelector(".folder-icon").textContent="📂";
        }
      }
    });
  }

  await loadPlans();
});
</script>

(function(){

  function injectTeamMeetings(){

    const menu =
      document.querySelector(".quick-menu") ||
      document.querySelector(".menu-dropdown") ||
      document.querySelector(".menu-panel") ||
      document.querySelector(".menu-links") ||
      document.querySelector(".dropdown-menu");

    if(!menu) return;

    if(document.getElementById("teamMeetingsMenuLink")) return;

    const recycleLink = Array.from(menu.querySelectorAll("a"))
      .find(a =>
        (a.textContent || "")
          .toLowerCase()
          .includes("recycle")
      );

    if(!recycleLink) return;

    const newLink = recycleLink.cloneNode(true);

    newLink.id = "teamMeetingsMenuLink";

    newLink.href = "/teacher/team-meetings.html";

    newLink.textContent = "Team Meetings";

    // REMOVE INLINE STYLES THAT CAUSED STATIC LOOK
    newLink.removeAttribute("style");

    // insert before recycle bin
    recycleLink.parentNode.insertBefore(newLink, recycleLink);

  }

  injectTeamMeetings();

  setInterval(injectTeamMeetings, 1500);

})();

const canvas = document.getElementById('canvas');

let lessonData = JSON.parse(localStorage.getItem('lessonDraft') || '[]');

function renderCanvas() {
  canvas.innerHTML = "";

  lessonData.forEach((block, index) => {
    const div = document.createElement('div');
    div.className = "block";
    div.dataset.index = index;

    div.innerHTML = 
      <div class="block-content"></div>
      <button class="delete-btn">Delete</button>
    ;

    canvas.appendChild(div);
  });
}

// EVENT DELEGATION (FIX DELETE)
canvas.addEventListener("click", (e) => {
  if (e.target.classList.contains("delete-btn")) {
    const index = e.target.parentElement.dataset.index;
    lessonData.splice(index, 1);
    saveDraft();
    renderCanvas();
  }
});

// SAVE (FIX PERSISTENCE)
function saveDraft() {
  localStorage.setItem("lessonDraft", JSON.stringify(lessonData));
}

// AUTO SAVE LIKE GOOGLE DOCS
setInterval(() => {
  saveDraft();
  console.log("Auto-saved...");
}, 2000);

// INIT
renderCanvas();

import {
  collection,
  addDoc,
  getDocs
} from "https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js";

import { db } from "/js/firebase-app.js";

let folders = [];
let lessons = [];
let builder = [];

async function load(){
  let f = await getDocs(collection(db,"lessonFolders"));
  f.forEach(d=>folders.push({id:d.id,...d.data()}));

  let l = await getDocs(collection(db,"lessons"));
  l.forEach(d=>lessons.push({id:d.id,...d.data()}));

  render();
}

function render(){
  document.getElementById("folders").innerHTML =
    folders.map(f=>`<div>${f.name}</div>`).join("");

  document.getElementById("lessons").innerHTML =
    lessons.map(l=>`<div>${l.title}</div>`).join("");
}

window.openFolder = async function(){
  let name = prompt("Folder name");

  await addDoc(collection(db,"lessonFolders"),{
    name
  });

  location.reload();
}

(function(){
  function load(src){
    return new Promise((resolve,reject)=>{
      if(document.querySelector('script[src="'+src+'"]')) return resolve();
      const s=document.createElement("script");
      s.src=src;
      s.onload=resolve;
      s.onerror=reject;
      document.head.appendChild(s);
    });
  }

  async function ready(){
    await load("https://www.gstatic.com/firebasejs/10.12.4/firebase-app-compat.js");
    await load("https://www.gstatic.com/firebasejs/10.12.4/firebase-firestore-compat.js");
    await load("https://www.gstatic.com/firebasejs/10.12.4/firebase-storage-compat.js");
    await load("/__/firebase/init.js");

    return {
      db: firebase.firestore(),
      storage: firebase.storage()
    };
  }

  function cleanKey(v){
    return String(v || "").trim().replace(/\s+/g,"").toUpperCase();
  }

  function makeId(prefix){
    return prefix + "-" + Date.now() + "-" + Math.random().toString(36).slice(2,8);
  }

  async function saveDoc(collectionName, data, id){
    const { db } = await ready();
    const docId = id || data.id || makeId(collectionName);

    await db.collection(collectionName).doc(docId).set({
      ...data,
      id: docId,
      updatedAt: new Date().toISOString()
    }, { merge:true });

    return docId;
  }

  async function getDocs(collectionName){
    const { db } = await ready();
    const snap = await db.collection(collectionName).get();
    const rows = [];
    snap.forEach(doc => rows.push({ id:doc.id, ...doc.data() }));
    return rows;
  }

  async function getDoc(collectionName, id){
    const { db } = await ready();
    const ref = await db.collection(collectionName).doc(id).get();
    if(!ref.exists) return null;
    return { id:ref.id, ...ref.data() };
  }

  async function deleteDoc(collectionName, id){
    const { db } = await ready();
    await db.collection(collectionName).doc(id).delete();
  }

  async function uploadFile(file, folder, subfolder){
    const { storage } = await ready();

    const safeName = file.name.replace(/[^\w.\-]+/g,"_");
    const safeFolder = String(folder || "uploads").replace(/[^\w\-\/]+/g,"_");
    const safeSub = subfolder ? String(subfolder).replace(/[^\w\-\/]+/g,"_") + "/" : "";
    const path = "impact/" + safeFolder + "/" + safeSub + Date.now() + "-" + safeName;

    const ref = storage.ref(path);
    await ref.put(file);

    return {
      url: await ref.getDownloadURL(),
      path,
      name: file.name,
      mimeType: file.type,
      size: file.size,
      folder: safeFolder,
      createdAt: new Date().toISOString()
    };
  }

  async function saveLessonPlan(plan){
    return await saveDoc("lessonPlans", {
      title: plan.title || plan.name || "Untitled Lesson Plan",
      name: plan.title || plan.name || "Untitled Lesson Plan",
      level: plan.level || "Absolute Beginner",
      content: plan.content || "",
      media: plan.media || [],
      createdAt: plan.createdAt || new Date().toISOString()
    }, plan.id);
  }

  async function getLessonPlans(){
    return await getDocs("lessonPlans");
  }

  async function getLessonPlan(id){
    return await getDoc("lessonPlans", id);
  }

  async function deleteLessonPlan(id){
    return await deleteDoc("lessonPlans", id);
  }

  async function validatePasskey(passkeyValue, gate){
    const { db } = await ready();
    const typed = cleanKey(passkeyValue);

    const snap = await db.collection("impactPasskeys").get();

    let foundDoc = null;
    let foundData = null;

    snap.forEach(doc=>{
      const data = doc.data();
      const stored = cleanKey(data.passkey || data.key || data.code);

      if(stored === typed){
        foundDoc = doc;
        foundData = data;
      }
    });

    if(!foundData){
      return { ok:false, message:"Invalid passkey." };
    }

    if(String(foundData.status || "active").toLowerCase() !== "active"){
      return { ok:false, message:"This passkey has been deactivated." };
    }

    await db.collection("impactGateVisits").add({
      passkeyId: foundDoc.id,
      name: foundData.name || "Unknown",
      role: foundData.role || "Teacher",
      gate: gate || "personnel",
      passkey: foundData.passkey || passkeyValue,
      time: new Date().toISOString()
    });

    return {
      ok:true,
      ownerName: foundData.name || "Personnel",
      role: foundData.role || "Teacher"
    };
  }

  async function createPasskey(item){
    const { db } = await ready();

    const existing = await db.collection("impactPasskeys")
      .where("passkey","==",item.passkey)
      .limit(1)
      .get();

    const data = {
      name:item.name || "Unnamed",
      role:item.role || "Teacher",
      passkey:item.passkey,
      status:item.status || "active",
      createdAt:item.createdAt || new Date().toISOString(),
      source:item.source || "configuration_page"
    };

    if(existing.empty){
      const ref = await db.collection("impactPasskeys").add(data);
      return ref.id;
    }

    await existing.docs[0].ref.set(data,{merge:true});
    return existing.docs[0].id;
  }

  window.ImpactFirebaseStore = {
    ready,
    saveDoc,
    getDocs,
    getDoc,
    deleteDoc,
    uploadFile,

    saveLessonPlan,
    getLessonPlans,
    getLessonPlan,
    deleteLessonPlan,

    validatePasskey,
    createPasskey,

    collections:{
      passkeys:"impactPasskeys",
      visits:"impactGateVisits",
      lessonPlans:"lessonPlans",
      learners:"learners",
      classes:"classes",
      recordings:"recordings",
      activities:"activities",
      progress:"progress",
      delfMocks:"delfMocks",
      configuration:"configuration"
    }
  };
})();
