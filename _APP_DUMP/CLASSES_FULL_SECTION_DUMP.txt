<script>
document.addEventListener("DOMContentLoaded", async () => {
  const db = firebase.firestore();

  const form = document.getElementById("classForm");
  const level = document.getElementById("level");
  const learner = document.getElementById("learner");
  const teacher = document.getElementById("teacher");
  const day = document.getElementById("day");
  const time = document.getElementById("time");
  const duration = document.getElementById("duration");
  const meetRoom = document.getElementById("meetRoom");
  const status = document.getElementById("status");
  const notes = document.getElementById("notes");
  const roomNote = document.getElementById("roomNote");
  const saveStatus = document.getElementById("saveStatus");
  const classList = document.getElementById("classList");
  const classCount = document.getElementById("classCount");

  let allClasses = [];
  let editingId = null;

  function activeKey(id){ return "impactActiveClass_" + id; }
  function getActive(id){
    try { return JSON.parse(localStorage.getItem(activeKey(id)) || "null"); }
    catch(e){ return null; }
  }
  function setActive(id, data){ localStorage.setItem(activeKey(id), JSON.stringify(data)); }
  function clearActive(id){ localStorage.removeItem(activeKey(id)); }

  function getRoomURL(){
    const option = meetRoom.options[meetRoom.selectedIndex];
    return option?.dataset?.url || "";
  }

  async function loadClasses(){
    classList.innerHTML = '<div class="empty">Loading classes...</div>';

    const snapshot = await db.collection("classes").orderBy("createdAt","desc").get();

    allClasses = snapshot.docs.map(doc => ({ id:doc.id, ...doc.data() }));

    renderClasses();
    refreshRoomAvailability();
  }

  function renderClasses(){
    classCount.textContent = allClasses.length + (allClasses.length === 1 ? " class" : " classes");

    if(!allClasses.length){
      classList.innerHTML = '<div class="empty">No classes scheduled yet.</div>';
      return;
    }

    classList.innerHTML = "";

    allClasses.forEach(item => {
      const active = getActive(item.id);
      const card = document.createElement("div");
      card.className = "class-card";
      card.dataset.classId = item.id;

      card.innerHTML = `
        <div>
          <h3>${item.learner || "Unnamed Learner"} — ${item.level || "No level"}</h3>
          <div class="class-meta">${item.day || ""} · ${item.time || ""} · ${item.duration || ""}</div>
          <div class="class-meta">${item.teacher || ""} · ${item.meetRoom || ""} · ${item.status || ""}</div>
          ${active ? '<div class="class-meta" style="color:#d9f99d;font-weight:900;">Class in progress — started ' + (active.startedAtText || "") + '</div>' : ""}
        </div>

        <div class="class-actions">
          <button type="button" class="open-meet-btn ${active ? "running" : ""}">
            ${active ? "End Class" : "Open Meet"}
          </button>
          <button type="button" data-edit="${item.id}">Edit</button>
          <button type="button" data-delete="${item.id}">Delete</button>
        </div>
      `;

      card.querySelector(".open-meet-btn").onclick = () => handleClassStartEnd(item, card);
      card.querySelector("[data-edit]").onclick = () => editClass(item);
      card.querySelector("[data-delete]").onclick = () => deleteClass(item);

      classList.appendChild(card);
    });
  }

  async function handleClassStartEnd(item, card){
    const active = getActive(item.id);

    if(!active){
      if(!item.meetUrl){
        alert("No Meet link saved.");
        return;
      }

      const now = new Date();

      const logRef = await db.collection("classTracker").add({
        classId:item.id,
        learner:item.learner || "",
        teacher:item.teacher || "",
        level:item.level || "",
        day:item.day || "",
        time:item.time || "",
        duration:item.duration || "",
        meetRoom:item.meetRoom || "",
        meetUrl:item.meetUrl || "",
        startedAt:now.toISOString(),
        startedAtText:now.toLocaleString(),
        endedAt:"",
        endedAtText:"",
        minutesSpent:0,
        earlyEnd:false,
        earlyEndReason:"",
        earlyEndRemark:"",
        status:"Started"
      });

      setActive(item.id, {
        logId:logRef.id,
        startedAt:now.toISOString(),
        startedAtText:now.toLocaleString()
      });

      window.open(item.meetUrl, "_blank");
      renderClasses();
      return;
    }

    const startedAt = new Date(active.startedAt);
    const endedAt = new Date();
    const minutesSpent = Math.round((endedAt - startedAt) / 60000);

    if(minutesSpent < 40){
      showEarlyBox(item, card, active, minutesSpent);
      return;
    }

    await finishClass(item, active, minutesSpent, "", "");
    clearActive(item.id);
    renderClasses();
  }

  function showEarlyBox(item, card, active, minutesSpent){
    const old = card.querySelector(".early-box");
    if(old) old.remove();

    const box = document.createElement("div");
    box.className = "early-box";
    box.innerHTML = `
      <h4>Class ended before 40 minutes.</h4>
      <label>Reason</label>
      <select class="early-reason">
        <option value="">Select reason</option>
        <option>Learner did not join</option>
        <option>Internet issue</option>
        <option>Power outage</option>
        <option>Technical issue</option>
        <option>Teacher emergency</option>
        <option>Learner emergency</option>
        <option>Rescheduled</option>
        <option>Other</option>
      </select>
      <textarea class="early-remark" placeholder="Add remark if needed..."></textarea>
      <button type="button" class="early-save">Save Reason & End Class</button>
    `;

    card.appendChild(box);

    box.querySelector(".early-save").onclick = async () => {
      const reason = box.querySelector(".early-reason").value;
      const remark = box.querySelector(".early-remark").value.trim();

      if(!reason){
        alert("Please select a reason.");
        return;
      }

      await finishClass(item, active, minutesSpent, reason, remark);
      clearActive(item.id);
      renderClasses();
    };
  }

  async function finishClass(item, active, minutesSpent, reason, remark){
    const endedAt = new Date();

    await db.collection("classTracker").doc(active.logId).set({
      endedAt:endedAt.toISOString(),
      endedAtText:endedAt.toLocaleString(),
      minutesSpent,
      earlyEnd:minutesSpent < 40,
      earlyEndReason:reason || "",
      earlyEndRemark:remark || "",
      status:"Ended"
    }, { merge:true });

    await db.collection("classes").doc(item.id).set({
      lastClassStatus:minutesSpent < 40 ? "Ended Early" : "Completed",
      lastClassStartedAt:active.startedAt,
      lastClassEndedAt:endedAt.toISOString(),
      lastClassMinutes:minutesSpent,
      lastEarlyEndReason:reason || "",
      lastEarlyEndRemark:remark || ""
    }, { merge:true });
  }

  function editClass(item){
    editingId = item.id;
    level.value = item.level || "";
    learner.value = item.learner || "";
    teacher.value = item.teacher || "";
    day.value = item.day || "";
    time.value = item.time || "";
    duration.value = item.duration || "";
    meetRoom.value = item.meetRoom || "";
    status.value = item.status || "Scheduled";
    notes.value = item.notes || "";
    saveStatus.textContent = "Editing selected class";
  }

  async function deleteClass(item){
    if(!confirm("Delete this scheduled class?")) return;
    clearActive(item.id);
    await db.collection("classes").doc(item.id).delete();
    await loadClasses();
  }

  function refreshRoomAvailability(){
    const selectedDay = day.value;
    const selectedTime = time.value;

    Array.from(meetRoom.options).forEach(option => {
      if(!option.value) return;

      option.disabled = false;
      option.textContent = option.value;

      if(!selectedDay || !selectedTime) return;

      const unavailable = allClasses.some(classItem =>
        classItem.id !== editingId &&
        classItem.day === selectedDay &&
        classItem.time === selectedTime &&
        classItem.meetRoom === option.value &&
        classItem.status !== "Cancelled"
      );

      if(unavailable){
        option.disabled = true;
        option.textContent = option.value + " — unavailable at selected day and time";
      }
    });

    roomNote.textContent = selectedDay && selectedTime
      ? "Unavailable rooms are greyed out for the selected day and time."
      : "Select a day and time to see unavailable meet rooms.";
  }

  async function saveClass(event){
    event.preventDefault();

    const selectedOption = meetRoom.options[meetRoom.selectedIndex];

    if(!selectedOption || !meetRoom.value){
      saveStatus.textContent = "Please select a Google Meet room.";
      return;
    }

    if(selectedOption.disabled){
      saveStatus.textContent = "This room is unavailable at the selected day and time.";
      return;
    }

    const now = new Date();

    const data = {
      level:level.value.trim(),
      learner:learner.value.trim(),
      teacher:teacher.value.trim(),
      day:day.value,
      time:time.value,
      duration:duration.value.trim(),
      meetRoom:meetRoom.value,
      meetUrl:getRoomURL(),
      status:status.value,
      notes:notes.value.trim(),
      updatedAt:now.toISOString(),
      updatedAtText:now.toLocaleString()
    };

    if(editingId){
      await db.collection("classes").doc(editingId).set(data,{ merge:true });
    }else{
      await db.collection("classes").add({
        ...data,
        createdAt:now.toISOString(),
        createdAtText:now.toLocaleString()
      });
    }

    form.reset();
    editingId = null;
    saveStatus.textContent = "Saved ✓";
    await loadClasses();

    setTimeout(() => saveStatus.textContent = "", 1400);
  }

  form.addEventListener("submit", saveClass);
  day.addEventListener("change", refreshRoomAvailability);
  time.addEventListener("change", refreshRoomAvailability);
  meetRoom.addEventListener("focus", refreshRoomAvailability);

  await loadClasses();
});
</script>

<script>
document.addEventListener("DOMContentLoaded",async()=>{
const db=firebase.firestore();
const wrap=document.getElementById("wrap");
const search=document.getElementById("search");
let allLogs=[];
let knownLearners=[];

const BILLABLE_MINUTES_THRESHOLD=20;
const FIRST_BILLING_START=new Date(2026,5,4); // 04 June 2026 — fixed for testing

function clean(v){return String(v||"").replace(/\s*minute\(s\)\s*$/i,"").trim()}
function ms(x){return new Date(x.startedAt||x.createdAt||x.loggedAt||0).getTime()||0}

function isBillable(x){
  if(typeof x.billable==="boolean") return x.billable;
  const canceled=String(x.lessonCanceled||"").trim().toLowerCase()==="yes";
  if(canceled) return false;
  const completedField=String(x.lessonCompleted||"").trim().toLowerCase();
  if(completedField==="no") return false;
  if(x.status==="Completed") return true;
  if(x.status==="Ended"){
    const mins=Number(x.minutesSpent||x.durationMinutes||0);
    return mins>=BILLABLE_MINUTES_THRESHOLD;
  }
  return false;
}

function resolveLearnerName(typedName){
  const typedNorm=typedName.trim().toLowerCase().replace(/\s+/g," ");
  for(const known of knownLearners){
    if(known.trim().toLowerCase().replace(/\s+/g," ")===typedNorm) return known;
  }
  return typedName.trim();
}

function formatDateLong(d){
  const day=String(d.getDate()).padStart(2,"0");
  const month=d.toLocaleString("en-GB",{month:"long"});
  const year=d.getFullYear();
  return `${day} ${month} ${year}`;
}

async function load(){
  const snap=await db.collection("classTracker").get();
  const seen=new Set();
  allLogs=snap.docs.map(d=>({id:d.id,...d.data()}))
    .sort((a,b)=>ms(a)-ms(b))
    .filter(x=>{
      const key=[x.learner,x.startedAt||x.startedAtText,x.endedAt||x.endedAtText,x.meetRoom].join("|");
      if(seen.has(key)) return false;
      seen.add(key);
      return true;
    });

  const nameSet=new Set();
  allLogs.forEach(x=>{if(x.learner) nameSet.add(x.learner);});
  knownLearners=Array.from(nameSet).sort();

  const datalist=document.getElementById("m_learner_list");
  if(datalist){
    datalist.innerHTML=knownLearners.map(n=>`<option value="${n.replace(/"/g,"&quot;")}">`).join("");
  }

  render(allLogs);
}

function render(logs){
  if(!logs.length){wrap.innerHTML='<div class="empty">No tracker records found.</div>';return}

  const groups={};
  logs.forEach(x=>{
    const name=x.learner||"Unnamed Learner";
    if(!groups[name]) groups[name]=[];
    groups[name].push(x);
  });

  wrap.innerHTML="";

  Object.keys(groups).sort().forEach(name=>{
    const sec=document.createElement("section");
    sec.className="learner";
    sec.innerHTML=`<h2>${name}</h2>`;

    let billableCount=0;
    groups[name].sort((a,b)=>ms(a)-ms(b)).forEach((x)=>{
      const billable=isBillable(x);
      if(billable) billableCount++;
      const session=billable?(billableCount+"/4"):"-/4";
      const entryType=(x.entryType||"auto").toUpperCase();
      const card=document.createElement("article");
      card.className="card";
      card.innerHTML=`
        <div class="pill">${session}</div>
        <div class="info">
          <div class="top">Level: ${x.level||""} · Session ${session} · Teacher: ${x.teacher||""} · Room: ${x.meetRoom||x.platform||""}</div>
          <div>Started: ${x.startedAtText||x.startedAt||""}</div>
          <div>Ended: ${x.endedAtText||x.endedAt||"Not ended"}</div>
          <div>Duration: ${x.minutesSpent||x.durationMinutes||0} minute(s)</div>
          ${x.lessonCompleted?`<div>Lesson completed: ${clean(x.lessonCompleted)}</div>`:""}
          ${x.lessonCanceled?`<div>Lesson canceled: ${clean(x.lessonCanceled)}</div>`:""}
          ${x.studentParticipation?`<div>Student participation: ${clean(x.studentParticipation)}</div>`:""}
          ${x.teacherRemarks?`<div>Remarks: ${clean(x.teacherRemarks)}</div>`:""}
          <span class="status">${x.status||"Logged"}</span>
          <span class="badge badge-type">${entryType}</span>
          <span class="badge ${billable?"badge-billable":"badge-nonbillable"}">${billable?"BILLABLE":"NON-BILLABLE"}</span>
        </div>
        <button class="delete">Delete</button>
      `;
      card.querySelector(".delete").onclick=async()=>{
        if(confirm("Delete this tracker record?")){
          await db.collection("classTracker").doc(x.id).delete();
          await load();
        }
      };
      sec.appendChild(card);
    });

    wrap.appendChild(sec);
  });
}

search.oninput=()=>render(allLogs.filter(x=>JSON.stringify(x).toLowerCase().includes(search.value.toLowerCase())));

document.getElementById("pdfBtn").onclick=async()=>{
  const pdfBtn=document.getElementById("pdfBtn");
  pdfBtn.disabled=true;

  try{
    const historyRef=db.collection("billingExportHistory").doc("lastExport");
    const today=new Date();

    // PRODUCTION
const historySnap = await historyRef.get();
let rangeStart;

if(historySnap.exists && historySnap.data().lastExportEndDate){
    const lastEnd = new Date(historySnap.data().lastExportEndDate);
    rangeStart = new Date(lastEnd);
    rangeStart.setDate(rangeStart.getDate() + 1);
}else{
    rangeStart = new Date(FIRST_BILLING_START);
}

    

    const rangeStartLabel=formatDateLong(rangeStart);
    const rangeEndLabel=formatDateLong(today);
    const filename=`Billing Tracker from ${rangeStartLabel} to ${rangeEndLabel}.pdf`;

    const {jsPDF}=window.jspdf;
    const pdf=new jsPDF({orientation:"landscape",unit:"pt",format:"a4"});

    const margin=45;
    const top=80;
    const noW=42;
    const studentW=120;
    const dateW=62;
    const dateCols=10;
    const headerH=54;
    const rowH=82;
    const xNo=margin;
    const xStudent=xNo+noW;
    const xDates=xStudent+studentW;

    pdf.setFont("helvetica","bold");
    pdf.setFontSize(16);
    pdf.text(filename.replace(".pdf",""),margin,55);

    function cell(x,y,w,h,fill){
      pdf.setFillColor(fill[0],fill[1],fill[2]);
      pdf.setDrawColor(80,45,35);
      pdf.setLineWidth(.6);
      pdf.rect(x,y,w,h,"FD");
    }

    function centerText(txt,x,y,w,h,size){
      pdf.setFont("helvetica","bold");
      pdf.setFontSize(size);
      pdf.setTextColor(15,22,33);
      const lines=pdf.splitTextToSize(String(txt),w-8);
      const lineH=size+2;
      const total=lines.length*lineH;
      let yy=y+(h-total)/2+size;
      lines.forEach(line=>{
        pdf.text(line,x+w/2,yy,{align:"center"});
        yy+=lineH;
      });
    }

    cell(xNo,top,noW,headerH,[213,226,232]);
    cell(xStudent,top,studentW,headerH,[213,226,232]);
    cell(xDates,top,dateW*dateCols,headerH,[213,226,232]);
    centerText("NO.",xNo,top,noW,headerH,10);
    centerText("STUDENT",xStudent,top,studentW,headerH,10);
    centerText("CLASS DATES",xDates,top,dateW*dateCols,headerH,11);

    const billableLogs=allLogs.filter(x=>isBillable(x));
    const grouped={};
    billableLogs.forEach(x=>{
      const name=x.learner||"Student";
      const raw=x.startedAtText||x.startedAt||"";
      const date=String(raw).split(",")[0].trim();
      if(!date) return;
      if(!grouped[name]) grouped[name]=[];
      if(!grouped[name].includes(date)) grouped[name].push(date);
    });

    let y=top+headerH;
    let n=1;

    Object.keys(grouped).sort().forEach(name=>{
      if(y+rowH>540){
        pdf.addPage("a4","landscape");
        y=60;
        cell(xNo,y,noW,headerH,[213,226,232]);
        cell(xStudent,y,studentW,headerH,[213,226,232]);
        cell(xDates,y,dateW*dateCols,headerH,[213,226,232]);
        centerText("NO.",xNo,y,noW,headerH,10);
        centerText("STUDENT",xStudent,y,studentW,headerH,10);
        centerText("CLASS DATES",xDates,y,dateW*dateCols,headerH,11);
        y+=headerH;
      }

      const boxes=[];
      for(let i=0;i<grouped[name].length;i+=4){
        boxes.push(grouped[name].slice(i,i+4));
      }
      while(boxes.length<dateCols) boxes.push([]);

      cell(xNo,y,noW,rowH,[235,242,245]);
      cell(xStudent,y,studentW,rowH,[235,242,245]);

      pdf.setTextColor(20,25,35);
      pdf.setFont("helvetica","bold");
      pdf.setFontSize(9);
      pdf.text(String(n++),xNo+noW/2,y+18,{align:"center"});
      pdf.text(name,xStudent+10,y+18);

      boxes.slice(0,dateCols).forEach((dates,i)=>{
        const x=xDates+(i*dateW);
        cell(x,y,dateW,rowH,[235,242,245]);
        pdf.setFont("helvetica","normal");
        pdf.setFontSize(8);
        pdf.setTextColor(20,25,35);
        let yy=y+16;
        dates.forEach(d=>{
          pdf.text(d,x+6,yy);
          yy+=12;
        });
      });

      y+=rowH;
    });

    pdf.save(filename);

    await historyRef.set({
      lastExportEndDate:"2026-07-06T00:00:00.000Z",
      lastExportEndDateLabel:"06 July 2026",
      lastExportStartDateLabel:rangeStartLabel,
      lastExportFilename:filename,
      updatedAt:today.toISOString()
    });

  }catch(err){
    console.error(err);
    alert("Could not generate the Billing Tracker PDF. Please try again.");
  }finally{
    pdfBtn.disabled=false;
  }
};

  const learnerTyped=document.getElementById("m_learner").value.trim();
  const teacher=document.getElementById("m_teacher").value.trim();

  function fieldByLabel(labelText){
    const labels = Array.from(document.querySelectorAll("label"));
    const label = labels.find(l => (l.textContent || "").trim().toLowerCase() === labelText.toLowerCase());
    if(!label) return null;
    const wrap = label.closest("div") || label.parentElement;
    return wrap ? wrap.querySelector("input, select, textarea") : null;
  }

  function findButton(text){
    return Array.from(document.querySelectorAll("button")).find(b => (b.textContent || "").trim().toLowerCase() === text.toLowerCase());
  }

  function init(){
    if(!window.firebase || !firebase.firestore){
      setTimeout(init, 700);
      return;
    }

    const db = firebase.firestore();

    const level = fieldByLabel("Level");
    const learner = fieldByLabel("Learner");
    const teacher = fieldByLabel("Teacher");
    const date = fieldByLabel("Date");
    const day = fieldByLabel("Day");
    const time = fieldByLabel("Time");
    const duration = fieldByLabel("Duration");
    const meetRoom = fieldByLabel("Google Meet Room");
    const status = fieldByLabel("Status");
    const notes = fieldByLabel("Class Notes");
    const saveBtn = findButton("Save Class");

    if(!saveBtn || !meetRoom) return;

    let currentClassId = new URLSearchParams(location.search).get("edit") || "";

    function selectedDateKey(){
      return (date && date.value) ? date.value : ((day && day.value) ? day.value : "");
    }

    function cleanRoomText(text){
      return String(text || "").replace(/\s*—\s*unavailable.*$/i,"").trim();
    }

    function showRoomHint(message){
      let hint = document.getElementById("meetRoomAvailabilityHint");
      if(!hint){
        hint = document.createElement("div");
        hint.id = "meetRoomAvailabilityHint";
        hint.style.cssText = "margin-top:7px;font-size:.72rem;font-weight:800;color:rgba(255,255,255,.48);line-height:1.35;";
        meetRoom.insertAdjacentElement("afterend", hint);
      }
      hint.textContent = message || "";
    }

    async function getClasses(){
      const snap = await db.collection("classes").get();
      return snap.docs.map(d => ({ id:d.id, ...d.data() }));
    }

    async function refreshRoomAvailability(){
      const dateKey = selectedDateKey();
      const timeKey = time ? time.value : "";

      if(!dateKey || !timeKey){
        showRoomHint("Select date/day and time first to check room availability.");
        return;
      }

      const classes = await getClasses();

      Array.from(meetRoom.options).forEach(opt => {
        const roomName = cleanRoomText(opt.textContent || opt.value);
        opt.disabled = false;
        opt.textContent = roomName;

        const taken = classes.some(c =>
          c.id !== currentClassId &&
          String(c.dateKey || c.date || c.day || "") === String(dateKey) &&
          String(c.time || "") === String(timeKey) &&
          String(c.meetRoom || c.googleMeetRoom || "") === String(roomName) &&
          String(c.status || "").toLowerCase() !== "cancelled"
        );

        if(taken){
          opt.disabled = true;
          opt.textContent = roomName + " — unavailable at selected time";
        }
      });

      showRoomHint("Unavailable rooms are greyed out for the selected date/day and time.");
    }

    async function saveClass(e){
      e.preventDefault();
      e.stopPropagation();

      const roomName = meetRoom.options[meetRoom.selectedIndex] ? cleanRoomText(meetRoom.options[meetRoom.selectedIndex].textContent) : meetRoom.value;
      const dateKey = selectedDateKey();
      const timeKey = time ? time.value : "";

      await refreshRoomAvailability();

      if(meetRoom.options[meetRoom.selectedIndex] && meetRoom.options[meetRoom.selectedIndex].disabled){
        alert("This Google Meet room is unavailable at the selected date/day and time.");
        return;
      }

      const record = {
        level: level ? level.value : "",
        learner: learner ? learner.value : "",
        teacher: teacher ? teacher.value : "",
        date: date ? date.value : "",
        day: day ? day.value : "",
        dateKey,
        time: timeKey,
        duration: duration ? duration.value : "",
        meetRoom: roomName,
        googleMeetRoom: roomName,
        status: status ? status.value : "Scheduled",
        notes: notes ? notes.value : "",
        updatedAt: new Date().toISOString()
      };

      if(currentClassId){
        await db.collection("classes").doc(currentClassId).set(record,{merge:true});
      }else{
        const ref = await db.collection("classes").add({
          ...record,
          createdAt: new Date().toISOString()
        });
        currentClassId = ref.id;
        history.replaceState(null,"","/teacher/classes.html?edit=" + currentClassId);
      }

      saveBtn.textContent = "Saved ✓";
      setTimeout(()=>saveBtn.textContent = "Save Class", 1400);
      await refreshRoomAvailability();
    }

    saveBtn.onclick = null;
    saveBtn.addEventListener("click", saveClass, true);

    [date, day, time].forEach(el => {
      if(el){
        el.addEventListener("change", refreshRoomAvailability);
        el.addEventListener("input", refreshRoomAvailability);
      }
    });

    meetRoom.addEventListener("focus", refreshRoomAvailability);
    meetRoom.addEventListener("click", refreshRoomAvailability);

    refreshRoomAvailability();
  }

  if(document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded", init);
  }else{
    init();
  }
})();

function docRef(){
  return db.collection("academicCalendars").doc(String(currentYear));
}

function normalizeCalendarData(){
  if(!data.months) data.months = {};

  monthNames.forEach(month => {
    if(!Array.isArray(data.months[month])) data.months[month] = [];

    data.months[month] = data.months[month].map(item => {
      if(Array.isArray(item) && item.length >= 4){
        return [item[0], item[2], null];
      }

      if(Array.isArray(item) && item.length === 3){
        if(typeof item[2] === "boolean" || item[2] === null){
          return [item[0], item[1], item[2]];
        }

        return [item[0], item[2], null];
      }

      if(Array.isArray(item) && item.length === 2){
        return [item[0], item[1], null];
      }

      return ["", String(item || ""), null];
    });
  });
}

async function load(){
  yearLabel.textContent = currentYear;
  statusEl.textContent = "Loading calendar...";

  const fallback = () => {
    data = JSON.parse(JSON.stringify({ months: starter, notes:"" }));
    normalizeCalendarData();
    notesEl.value = data.notes || "";
    render();
    statusEl.textContent = "Calendar loaded offline. Firestore will sync when available.";
  };

  try{
    const timeout = new Promise((_, reject) =>
      setTimeout(() => reject(new Error("Firestore timeout")), 4500)
    );

    const snap = await Promise.race([docRef().get(), timeout]);

    if(snap.exists){
      data = snap.data();
    }else{
      data = JSON.parse(JSON.stringify({ months: starter, notes:"" }));
      await docRef().set(data,{merge:true});
    }

    normalizeCalendarData();
    notesEl.value = data.notes || "";
    render();

    statusEl.textContent = "Saved in Firestore";
  }catch(err){
    console.warn("Firestore load failed:", err);
    fallback();
  }
}

async function save(){
  normalizeCalendarData();
  data.notes = notesEl.value;

  try{
    await docRef().set(data,{merge:true});
    statusEl.textContent = "Saved ✓";
    setTimeout(()=>statusEl.textContent="Saved in Firestore",1000);
  }catch(err){
    console.warn("Firestore save failed:", err);
    localStorage.setItem("impactAcademicCalendar_" + currentYear, JSON.stringify(data));
    statusEl.textContent = "Saved locally. Firestore not reachable.";
  }
}

function makeStatus(month,index,item,row){
  const wrap = document.createElement("div");
  wrap.className = "status-actions";

  const yes = document.createElement("button");
  yes.className = "status-btn ok" + (item[2] === true ? " active" : "");
  yes.type = "button";

  const no = document.createElement("button");
  no.className = "status-btn no" + (item[2] === false ? " active" : "");
  no.type = "button";

  yes.onclick = async () => {
    data.months[month][index][2] = true;
    row.classList.add("done");
    yes.classList.add("active");
    no.classList.remove("active");
    await save();
  };

  no.onclick = async () => {
    data.months[month][index][2] = false;
    row.classList.remove("done");
    no.classList.add("active");
    yes.classList.remove("active");
    await save();
  };

  wrap.appendChild(yes);
  wrap.appendChild(no);

  return wrap;
}

function render(){
  normalizeCalendarData();
  monthsEl.innerHTML = "";

  monthNames.forEach(month => {
    const card = document.createElement("article");
    card.className = "card";

    const head = document.createElement("div");
    head.className = "card-head";
    head.innerHTML = '<span class="head-icon">▦</span>' + month;

    const list = document.createElement("div");
    list.className = "items";

    data.months[month].forEach((item,index) => {
      const row = document.createElement("div");
      row.className = "item";
      row.classList.toggle("done", item[2] === true);

      const date = document.createElement("div");
      date.className = "date";
      date.textContent = item[0] || "";

      const text = document.createElement("div");
      text.className = "event-text";
      text.textContent = item[1] || "";

      const status = makeStatus(month,index,item,row);

      const del = document.createElement("button");
      del.className = "delete";
      del.type = "button";
      del.textContent = "×";
      del.onclick = async () => {
        data.months[month].splice(index,1);
        render();
        await save();
      };

      row.appendChild(date);
      row.appendChild(text);
      row.appendChild(status);
      row.appendChild(del);
      list.appendChild(row);
    });

    const add = document.createElement("div");
    add.className = "add-line";
    add.innerHTML = '<input placeholder="Date"><input placeholder="Event"><button type="button">+</button>';

    add.querySelector("button").onclick = async () => {
      const inputs = add.querySelectorAll("input");
      const date = inputs[0].value.trim();
      const event = inputs[1].value.trim();

      if(!date || !event) return;

      data.months[month].push([date,event,null]);

      inputs[0].value = "";
      inputs[1].value = "";

      render();
      await save();
    };

    card.appendChild(head);
    card.appendChild(list);
    card.appendChild(add);
    monthsEl.appendChild(card);
  });
}
