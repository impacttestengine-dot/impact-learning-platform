(function(){
  function load(src){
    return new Promise((resolve,reject)=>{
      if(document.querySelector('script[src="'+src+'"]')) return resolve();
      const s=document.createElement("script");
      s.src=src;
      s.onload=resolve;
      s.onerror=reject;
      document.head.appendChild(s);
    });
  }

  async function ready(){
    await load("https://www.gstatic.com/firebasejs/10.12.4/firebase-app-compat.js");
    await load("https://www.gstatic.com/firebasejs/10.12.4/firebase-firestore-compat.js");
    await load("https://www.gstatic.com/firebasejs/10.12.4/firebase-storage-compat.js");
    await load("/__/firebase/init.js");

    return {
      db: firebase.firestore(),
      storage: firebase.storage()
    };
  }

  function cleanKey(v){
    return String(v || "").trim().replace(/\s+/g,"").toUpperCase();
  }

  function makeId(prefix){
    return prefix + "-" + Date.now() + "-" + Math.random().toString(36).slice(2,8);
  }

  async function saveDoc(collectionName, data, id){
    const { db } = await ready();
    const docId = id || data.id || makeId(collectionName);

    await db.collection(collectionName).doc(docId).set({
      ...data,
      id: docId,
      updatedAt: new Date().toISOString()
    }, { merge:true });

    return docId;
  }

  async function getDocs(collectionName){
    const { db } = await ready();
    const snap = await db.collection(collectionName).get();
    const rows = [];
    snap.forEach(doc => rows.push({ id:doc.id, ...doc.data() }));
    return rows;
  }

  async function getDoc(collectionName, id){
    const { db } = await ready();
    const ref = await db.collection(collectionName).doc(id).get();
    if(!ref.exists) return null;
    return { id:ref.id, ...ref.data() };
  }

  async function deleteDoc(collectionName, id){
    const { db } = await ready();
    await db.collection(collectionName).doc(id).delete();
  }

  async function uploadFile(file, folder, subfolder){
    const { storage } = await ready();

    const safeName = file.name.replace(/[^\w.\-]+/g,"_");
    const safeFolder = String(folder || "uploads").replace(/[^\w\-\/]+/g,"_");
    const safeSub = subfolder ? String(subfolder).replace(/[^\w\-\/]+/g,"_") + "/" : "";
    const path = "impact/" + safeFolder + "/" + safeSub + Date.now() + "-" + safeName;

    const ref = storage.ref(path);
    await ref.put(file);

    return {
      url: await ref.getDownloadURL(),
      path,
      name: file.name,
      mimeType: file.type,
      size: file.size,
      folder: safeFolder,
      createdAt: new Date().toISOString()
    };
  }

    }, plan.id);
  }

  async function validatePasskey(passkeyValue, gate){
    const { db } = await ready();
    const typed = cleanKey(passkeyValue);

    const snap = await db.collection("impactPasskeys").get();

    let foundDoc = null;
    let foundData = null;

    snap.forEach(doc=>{
      const data = doc.data();
      const stored = cleanKey(data.passkey || data.key || data.code);

      if(stored === typed){
        foundDoc = doc;
        foundData = data;
      }
    });

    if(!foundData){
      return { ok:false, message:"Invalid passkey." };
    }

    if(String(foundData.status || "active").toLowerCase() !== "active"){
      return { ok:false, message:"This passkey has been deactivated." };
    }

    await db.collection("impactGateVisits").add({
      passkeyId: foundDoc.id,
      name: foundData.name || "Unknown",
      role: foundData.role || "Teacher",
      gate: gate || "personnel",
      passkey: foundData.passkey || passkeyValue,
      time: new Date().toISOString()
    });

    return {
      ok:true,
      ownerName: foundData.name || "Personnel",
      role: foundData.role || "Teacher",
      level: foundData.level || ""
    };
  }

  async function createPasskey(item){
    const { db } = await ready();

    const existing = await db.collection("impactPasskeys")
      .where("passkey","==",item.passkey)
      .limit(1)
      .get();

    const data = {
      name:item.name || "Unnamed",
      role:item.role || "Teacher",
      level:item.level || "",
      passkey:item.passkey,
      status:item.status || "active",
      createdAt:item.createdAt || new Date().toISOString(),
      source:item.source || "configuration_page"
    };

    if(existing.empty){
      const ref = await db.collection("impactPasskeys").add(data);
      return ref.id;
    }

    await existing.docs[0].ref.set(data,{merge:true});
    return existing.docs[0].id;
  }

})();

        </div>`;

      // Insert after the row that was clicked

      const close = r=>{
        pop.remove();
        document.removeEventListener("click", outside);
        resolve(r);
      };

document.addEventListener("DOMContentLoaded", async function(){
  const db = firebase.firestore();
  const name  = sessionStorage.getItem("impactAccessName") || sessionStorage.getItem("impactLearnerName") || "";
  const level = sessionStorage.getItem("impactAccessLevel") || "";

  const pendingWrap = document.getElementById("pendingWrap");
  const foldersWrap = document.getElementById("foldersWrap");

  if(!name){
    pendingWrap.innerHTML = '<div class="empty-state"><h3>Please sign in</h3></div>';
    foldersWrap.innerHTML = "";
    return;
  }

  try{
    const [byLevel, byName, completedSnap] = await Promise.all([
      level ? db.collection("activityPlans").where("assignedLevels","array-contains",level).get() : Promise.resolve({docs:[]}),
      db.collection("activityPlans").where("assignedLearners","array-contains",name).get(),
      db.collection("learnerCompletions").where("learnerName","==",name).where("itemType","==","activity").get()
    ]);

    const completedIds = new Set();
    completedSnap.forEach(d=>completedIds.add(d.data().itemId));

    const itemMap = new Map();
    byLevel.docs.forEach(d=>itemMap.set(d.id, {id:d.id, ...d.data()}));
    byName.docs.forEach(d=>itemMap.set(d.id, {id:d.id, ...d.data()}));

    const allItems = Array.from(itemMap.values());
    const pending = allItems.filter(it=>!completedIds.has(it.id));
    const completed = allItems.filter(it=>completedIds.has(it.id));

    // ── PENDING ──
    if(!pending.length){
      pendingWrap.innerHTML = '<div class="empty-pending">✨ Nothing pending right now — you're all caught up!</div>';
    }else{
      pendingWrap.innerHTML = pending.map(it=>`
        <a class="pending-card" href="/teacher/=+it.id+">
          <div class="pending-icon">🎨</div>
          <div class="pending-info">
            <div class="pending-title">+(it.title||"Untitled")+</div>
            <div class="pending-meta">+(it.level?"Level "+it.level:"")+</div>
          </div>
          <div class="pending-arrow">→</div>
        </a>
      `).join("");
    }

    // ── COMPLETED — grouped into folders by level ──
    if(!completed.length){
      foldersWrap.innerHTML = '<div class="empty-state"><h3>No completed items yet</h3></div>';
    }else{
      const byLevelMap = {};
      completed.forEach(it=>{
        const lvl = it.level || "Other";
        if(!byLevelMap[lvl]) byLevelMap[lvl] = [];
        byLevelMap[lvl].push(it);
      });

      foldersWrap.innerHTML = '<div class="folders-grid" id="grid"></div>';
      const grid = document.getElementById("grid");
      let openFolder = null;

      Object.keys(byLevelMap).sort().forEach(lvl=>{
        const items = byLevelMap[lvl];
        const card = document.createElement("div");
        card.className = "folder-card";
        card.dataset.level = lvl;

        const rows = items.map(it=>`
          <a class="completed-row" href="/teacher/=+it.id+">
            <span class="completed-check">✓</span>+(it.title||"Untitled")+
          </a>
        `).join("");

        card.innerHTML = `
          <div class="folder-head">
            <span class="folder-icon">📁</span>
            <div class="folder-meta">
              <div class="folder-name">+lvl+</div>
              <div class="folder-sub">+items.length+ completed</div>
            </div>
            <span class="folder-chevron">›</span>
          </div>
          <div class="folder-body">+rows+</div>
        `;

        card.querySelector(".folder-head").addEventListener("click", function(){
          const wasOpen = card.classList.contains("open");
          document.querySelectorAll(".folder-card.open").forEach(c=>c.classList.remove("open"));
          if(!wasOpen) card.classList.add("open");
        });

        grid.appendChild(card);
      });
    }
  }catch(e){
    console.error(e);
    pendingWrap.innerHTML = '<div class="empty-state"><h3>Could not load items</h3></div>';
    foldersWrap.innerHTML = "";
  }
});
</script>
