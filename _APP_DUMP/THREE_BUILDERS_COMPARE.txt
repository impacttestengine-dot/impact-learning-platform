    if(!q){ learnerPickerList.classList.remove("show"); learnerPickerList.innerHTML=""; return; }
    const matches = allLearners.filter(l=>l.name.toLowerCase().includes(q) && !selectedLearners.includes(l.name));
    if(!matches.length){ learnerPickerList.classList.remove("show"); learnerPickerList.innerHTML=""; return; }
    learnerPickerList.innerHTML = matches.map(l=>`<div class="learner-picker-row" data-pick-learner="${l.name}">${l.name} <span style="opacity:.5">(${l.level||"no level"})</span></div>`).join("");
    learnerPickerList.classList.add("show");
  });

  learnerPickerList.addEventListener("click", function(e){
    const row = e.target.closest("[data-pick-learner]");
    if(!row) return;
    const name = row.dataset.pickLearner;
    if(!selectedLearners.includes(name)) selectedLearners.push(name);
    renderSelectedLearnerChips();
    learnerSearchInput.value = "";
    learnerPickerList.classList.remove("show");
    learnerPickerList.innerHTML = "";
  });

  assignLearnersSelected.addEventListener("click", function(e){
    const btn = e.target.closest("[data-remove-learner]");
    if(!btn) return;
    selectedLearners = selectedLearners.filter(n=>n!==btn.dataset.removeLearner);
    renderSelectedLearnerChips();
  });
}

// ── CREATE BLOCK ──────────────────────────────────────────
function createBlock(type, data){
  const tmpl = TEMPLATES[type];
  if(!tmpl) return null;

  const block = document.createElement("div");
  block.className = "c-block";
  block.setAttribute("data-type", type);
  block.setAttribute("draggable", "true");

  const actions = document.createElement("div");
  actions.className = "block-actions";
  actions.innerHTML = `
    <button class="act-btn" data-act="up" title="Move up">↑</button>
    <button class="act-btn" data-act="down" title="Move down">↓</button>
    <div class="act-divider"></div>
    <button class="act-btn" data-act="dupe" title="Duplicate">⧉</button>
    <div class="act-divider"></div>
    <button class="act-btn danger" data-act="del" title="Delete">✕</button>
  `;

  const inner = document.createElement("div");
  inner.className = "c-block-inner";
  inner.innerHTML = tmpl.render(data);

  block.appendChild(actions);
  block.appendChild(inner);

  bindBlock(block);
  return block;
}

function bindBlock(block){
  // ── Align / resize toolbar ──
  const toolbar = block.querySelector(".block-toolbar");
  if(toolbar){
    toolbar.addEventListener("click", function(e){
      e.stopPropagation();
      const alignBtn = e.target.closest("[data-align]");
      if(alignBtn){
        pushUndo();
        block.classList.remove("align-left","align-center","align-right");
        block.classList.add("align-"+alignBtn.dataset.align);
        toolbar.querySelectorAll("[data-align]").forEach(b=>b.classList.remove("active"));
        alignBtn.classList.add("active");
      }
      const sizeBtn = e.target.closest("[data-size]");
      if(sizeBtn){
        pushUndo();
        block.classList.remove("bw-small","bw-medium","bw-large");
        block.classList.add("bw-"+sizeBtn.dataset.size);
        toolbar.querySelectorAll("[data-size]").forEach(b=>b.classList.remove("active"));
        sizeBtn.classList.add("active");
      }
    });
  }

  // ── Media File block: click to upload, preview immediately ──
  const mediaEmpty   = block.querySelector(".cb-media-empty");
  const mediaInput   = block.querySelector(".cb-media-input");
  const mediaPreview = block.querySelector(".cb-media-preview");
  const mediaReplace = block.querySelector(".cb-media-replace");

  async function handleMediaFile(file){
    if(!file) return;
    const isVideo = file.type.startsWith("video/");

    // Show a local preview immediately via a blob URL (instant, no upload wait)
    const localPreviewUrl = URL.createObjectURL(file);
    mediaPreview.innerHTML = isVideo
      ? `<video src="${localPreviewUrl}" controls></video>`
      : `<img src="${localPreviewUrl}" alt="">`;
    mediaPreview.classList.add("show");
    if(mediaEmpty) mediaEmpty.style.display = "none";

    let statusEl = block.querySelector(".cb-media-status");
    if(!statusEl){
      statusEl = document.createElement("div");
      statusEl.className = "cb-media-status";
      statusEl.style.cssText = "margin-top:8px;font-size:.76rem;font-weight:700;color:#9ca3af";
      mediaPreview.after(statusEl);
    }
    statusEl.textContent = "Uploading...";

    try{
      const ext = file.name.split(".").pop() || (isVideo ? "mp4" : "jpg");
      const path = `lessonMedia/${Date.now()}_${Math.random().toString(36).slice(2,8)}.${ext}`;
      const snap = await storage.ref(path).put(file);
      const downloadUrl = await snap.ref.getDownloadURL();

      block.dataset.mediaSrc  = downloadUrl;
      block.dataset.mediaKind = isVideo ? "video" : "image";
      statusEl.textContent = "";
      statusEl.remove();

      let replaceBtn = block.querySelector(".cb-media-replace");
      if(!replaceBtn){
        replaceBtn = document.createElement("button");
        replaceBtn.type = "button";
        replaceBtn.className = "cb-media-replace";
        replaceBtn.textContent = "Replace file";
        mediaPreview.after(replaceBtn);
        replaceBtn.addEventListener("click", function(e){
          e.stopPropagation();
          mediaInput.click();
        });
      }
    }catch(err){
      console.error("Media upload failed:", err);
      statusEl.textContent = "⚠️ Upload failed — try again";
      statusEl.style.color = "#dc2626";
      toast("⚠️ Could not upload file: " + err.message);
    }
  }

  if(mediaEmpty && mediaInput){
    mediaEmpty.addEventListener("click", function(e){
      e.stopPropagation();
      mediaInput.click();
    });
    mediaInput.addEventListener("click", function(e){ e.stopPropagation(); });
    mediaInput.addEventListener("change", function(e){
      e.stopPropagation();
      handleMediaFile(mediaInput.files[0]);
    });
  }
  if(mediaReplace && mediaInput){
    mediaReplace.addEventListener("click", function(e){
      e.stopPropagation();
      mediaInput.click();
    });
  }
  block.addEventListener("click", function(e){
    if(e.target.closest(".block-actions")) return;
    e.stopPropagation();
    selectBlock(block);
  });

  const actionBar = block.querySelector(".block-actions");
  actionBar.addEventListener("click", function(e){
    e.stopPropagation();
    const btn = e.target.closest("[data-act]");
    if(!btn) return;
    const act = btn.dataset.act;

    if(act==="del"){
      pushUndo();
      block.remove();
      selectedBlock = null;
      showEmpty();
    }
    if(act==="up"){
      pushUndo();
      const prev = block.previousElementSibling;
      if(prev && prev.classList.contains("c-block")) canvas.insertBefore(block, prev);
    }
    if(act==="down"){
      pushUndo();
      const next = block.nextElementSibling;
      if(next && next.classList.contains("c-block")) canvas.insertBefore(next, block);
    }
    if(act==="dupe"){
      pushUndo();
      const type = block.dataset.type;
      const dupe = createBlock(type, extractData(block));
      block.after(dupe);
      selectBlock(dupe);
    }
  });

  // Add option button for multiple choice
  const mcAdd = block.querySelector(".mc-add");
  if(mcAdd){
    mcAdd.addEventListener("click", function(e){
      e.stopPropagation();
      pushUndo();
      const grid = block.querySelector(".mc-options");
      const opt = document.createElement("div");
      opt.className = "mc-option";
      opt.innerHTML = `<input type="radio" disabled><span class="mc-option-label" contenteditable="true">Option ${grid.children.length+1}</span>`;
      grid.appendChild(opt);
    });
  }

  // Wire up File Upload block: Browse Files button triggers hidden input,
  // selecting a file shows its name in the builder preview (preview only —
  // this is the builder canvas, not a live form; actual uploads happen
  // on the live-view.html page where learners fill out the form)
  const uploadBtn = block.querySelector(".impact-upload-btn");
  const hiddenFileInput = block.querySelector(".impact-hidden-upload");
  if(uploadBtn && hiddenFileInput){
    uploadBtn.addEventListener("click", function(e){
      e.stopPropagation();
      hiddenFileInput.click();
    });
    hiddenFileInput.addEventListener("click", function(e){ e.stopPropagation(); });
    hiddenFileInput.addEventListener("change", function(e){
      e.stopPropagation();
      const files = Array.from(hiddenFileInput.files);
      let nameEl = block.querySelector(".impact-upload-filename");
      if(!nameEl){
        nameEl = document.createElement("div");
        nameEl.className = "impact-upload-filename";
        hiddenFileInput.closest(".impact-drop-zone").appendChild(nameEl);
      }
      if(files.length){
        nameEl.textContent = files.length===1 ? "Selected: "+files[0].name : files.length+" files selected";
        nameEl.classList.add("show");
      }
    });
  }

  // Add image for picture choice
  const pcAdd = block.querySelector(".pc-add-img");
  if(pcAdd){
    pcAdd.addEventListener("click", function(e){
      e.stopPropagation();
      pcImageInput.click();
      pcImageInput.onchange = function(){
        const files = Array.from(pcImageInput.files);
        const grid = block.querySelector(".pc-grid");
        files.forEach(file=>{
          const reader = new FileReader();
          reader.onload = function(ev){
            const item = document.createElement("div");
            item.className = "pc-item";

            grid.appendChild(item);
          };
          reader.readAsDataURL(file);
        });
        pcImageInput.value="";
      };
    });
  }

  // Drag and drop reorder
  block.addEventListener("dragstart", function(e){
    dragSrc = block;
    e.dataTransfer.effectAllowed = "move";
    setTimeout(()=>block.style.opacity=".4", 0);
  });
  block.addEventListener("dragend", function(){
    block.style.opacity="";
    allBlocks().forEach(b=>b.classList.remove("drag-over"));
    dragSrc = null;
  });
  block.addEventListener("dragover", function(e){
    e.preventDefault();
    if(dragSrc && dragSrc !== block) block.classList.add("drag-over");
  });
  block.addEventListener("dragleave", function(){
    block.classList.remove("drag-over");
  });
  block.addEventListener("drop", function(e){
    e.preventDefault();
    block.classList.remove("drag-over");

    pushUndo();
    const rect = block.getBoundingClientRect();
    if(e.clientY < rect.top + rect.height/2) canvas.insertBefore(dragSrc, block);
    else block.after(dragSrc);
  });
}

// ── SERIALISE / DESERIALISE ───────────────────────────────
function extractData(block){
  const type = block.dataset.type;
  const data = { type };

  if(type==="header"||type==="subheader"||type==="text"){
    data.content = block.querySelector("[contenteditable]").innerHTML;
  }
  if(type==="pagebreak"){
    const wrap = block.querySelector(".page-break-wrap");
    data.pageNum = wrap ? parseInt(wrap.dataset.page)||2 : 2;
  }
  if(type==="shorttext"||type==="longtext"||type==="date"||type==="number"||type==="website"||type==="fileupload"||type==="signature"||type==="mediaanswer"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
  }
  if(type==="multiplechoice"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.options = Array.from(block.querySelectorAll(".mc-option-label")).map(el=>el.innerHTML);
  }
  if(type==="picturechoice"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.images = Array.from(block.querySelectorAll(".pc-img-placeholder img")).map(img=>img.src);
  }
  if(type==="dropdown"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.options = Array.from(block.querySelectorAll("select option")).slice(1).map(o=>o.textContent);
  }
    // Persist media file data URL + block styling so it survives save/reload
  if(type==="mediafile"){
    data.src  = block.dataset.mediaSrc  || "";
    data.kind = block.dataset.mediaKind || "";
  }
  const alignClass = ["align-left","align-center","align-right"].find(c=>block.classList.contains(c));
  const sizeClass   = ["bw-small","bw-medium","bw-large"].find(c=>block.classList.contains(c));
  data.align = alignClass ? alignClass.replace("align-","") : "center";
  data.size  = sizeClass  ? sizeClass.replace("bw-","")     : "large";
  return data;
}

function serializeBlocks(){
  return allBlocks().map(extractData);
}

function rebuildFromBlocks(blocks){
  allBlocks().forEach(b=>b.remove());
  selectedBlock = null;
  pageCount = 1;
  if(!blocks || !blocks.length){ showEmpty(); return; }
  hideEmpty();
  blocks.forEach(data=>{
    const block = createBlock(data.type, data);
    if(!block) return;
    canvas.appendChild(block);
    // Restore editable content
    if(data.content){
      const ce = block.querySelector("[contenteditable]");
      if(ce) ce.innerHTML = data.content;
    }
    if(data.label){
      const lel = block.querySelector(".q-label-text");
      if(lel) lel.innerHTML = data.label;
    }
    if(data.hint){
      const hel = block.querySelector(".q-hint");
      if(hel) hel.innerHTML = data.hint;
    }
  });
}

// ── SAVE ─────────────────────────────────────────────────
async function save(){
  try{
    const blocks = serializeBlocks();
    const payload = {
createdByTeacher:(sessionStorage.getItem("impactAccessName")||sessionStorage.getItem("impactPersonnelName")||`"),
      title: titleInput.value.trim() || "Untitled Lesson",
      level: levelSelect.value,
      blocks: blocks,
      version: 2,
      updatedAt: new Date().toISOString()
    };

    // Guard: warn (but still attempt) if payload is unexpectedly large —
    // Firestore documents are capped at ~1MB. Base64 media is no longer
    // stored inline (uploaded to Storage instead), so this should rarely
    // trigger, but it's a safety net against any future large-content block.
    const approxSize = JSON.stringify(payload).length;
    if(approxSize > 900000){
      toast("⚠️ This lesson is very large and may fail to save. Consider splitting it up.", 6000);
    }

    if(currentId){
      await db.collection("lessonPlans").doc(currentId).set(payload, {merge:true});
    } else {
      const ref = await db.collection("lessonPlans").add(payload);
      currentId = ref.id;
      history.replaceState(null,"","/teacher/lesson-builder.html?edit="+currentId);
    }
    tick();
    return currentId;
  }catch(err){
    console.error("Save failed:", err);
    toast("⚠️ Save failed: " + err.message, 6000);
    return null;
  }
}

function questionRowHtml(qid, data){
  const opts = (data && data.options) ? data.options : ["Option A","Option B","Option C","Option D"];
  const correctIdx = data && typeof data.correct === "number" ? data.correct : 0;
  const optsHtml = opts.map((o,i)=>`
    <label class="delf-answer-opt">
      <input type="radio" name="correct_${qid}" data-qid="${qid}" data-optidx="${i}" ${i===correctIdx?"checked":""}>
      <span contenteditable="true">${o}</span>
    </label>`).join("");
  return `
    <div class="delf-question-row" data-qid="${qid}">
      <button type="button" class="delf-remove-question" data-qid="${qid}">Remove</button>
      <div class="delf-question-text" contenteditable="true" data-ph="Type the question here...">${data&&data.question?data.question:""}</div>
      <div class="delf-answer-options">${optsHtml}</div>
    </div>`;
}

function createBlock(type, data){
  const tmpl = TEMPLATES[type];
  if(!tmpl) return null;
  const block = document.createElement("div");
  block.className = "c-block";
  block.setAttribute("data-type", type);
  const actions = document.createElement("div");
  actions.className = "block-actions";
  actions.innerHTML = `<button class="act-btn" data-act="up">↑</button><button class="act-btn" data-act="down">↓</button><div class="act-divider"></div><button class="act-btn" data-act="dupe">⧉</button><div class="act-divider"></div><button class="act-btn danger" data-act="del">✕</button>`;
  const inner = document.createElement("div");
  inner.className = "c-block-inner";
  inner.innerHTML = tmpl.render(data);
  block.appendChild(actions);
  block.appendChild(inner);
  bindBlock(block);
  return block;
}

function bindBlock(block){
  block.addEventListener("click", function(e){
    if(e.target.closest(".block-actions")) return;
    e.stopPropagation();
    selectBlock(block);
  });

  block.querySelector(".block-actions").addEventListener("click", function(e){
    e.stopPropagation();
    const btn = e.target.closest("[data-act]"); if(!btn) return;
    const act = btn.dataset.act;
    if(act==="del"){ pushUndo(); block.remove(); selectedBlock=null; showEmpty(); }
    if(act==="up"){ pushUndo(); const p=block.previousElementSibling; if(p&&p.classList.contains("c-block")) canvas.insertBefore(block,p); }
    if(act==="down"){ pushUndo(); const n=block.nextElementSibling; if(n&&n.classList.contains("c-block")) canvas.insertBefore(n,block); }
    if(act==="dupe"){ pushUndo(); const d=createBlock(block.dataset.type, extractData(block)); block.after(d); selectBlock(d); }
  });

  // ── Audio upload for Listening blocks ──
  const audioZone = block.querySelector(".delf-audio-zone");
  const audioInput = block.querySelector(".delf-audio-input");
  const audioPlayer = block.querySelector(".delf-audio-player");
  const audioStatus = block.querySelector(".delf-audio-status");

  if(audioZone && audioInput){
    audioZone.addEventListener("click", function(e){ e.stopPropagation(); audioInput.click(); });
    audioInput.addEventListener("click", function(e){ e.stopPropagation(); });
    audioInput.addEventListener("change", async function(e){
      e.stopPropagation();
      const file = audioInput.files[0];
      if(!file) return;
      audioStatus.textContent = "Uploading...";
      try{
        const path = `delfAudio/${Date.now()}_${Math.random().toString(36).slice(2,8)}.${file.name.split(".").pop()||"mp3"}`;
        const snap = await storage.ref(path).put(file);
        const url = await snap.ref.getDownloadURL();
        block.dataset.audioSrc = url;
        audioPlayer.innerHTML = `<audio src="${url}" controls></audio>`;
        audioPlayer.classList.add("show");
        audioZone.style.display = "none";
        audioStatus.textContent = "";
      }catch(err){
        audioStatus.textContent = "⚠️ Upload failed: " + err.message;
      }
    });
  }

  // ── Add / remove questions for Listening + Reading blocks ──
  const addQBtn = block.querySelector(".delf-add-question");
  if(addQBtn){
    addQBtn.addEventListener("click", function(e){
      e.stopPropagation();
      pushUndo();
      const wrap = block.querySelector(".delf-questions-wrap");
      const qid = "q"+Date.now()+"_"+Math.random().toString(36).slice(2,6);
      const div = document.createElement("div");
      div.innerHTML = questionRowHtml(qid, {question:"",options:["Option A","Option B","Option C","Option D"],correct:0});
      wrap.appendChild(div.firstElementChild);
    });
  }
  const qWrap = block.querySelector(".delf-questions-wrap");
  if(qWrap){
    qWrap.addEventListener("click", function(e){
      const rmBtn = e.target.closest(".delf-remove-question");
      if(rmBtn){
        e.stopPropagation();
        pushUndo();
        rmBtn.closest(".delf-question-row").remove();
      }
    });
  }
}

function extractData(block){
  const type = block.dataset.type;
  const data = { type };

  if(type==="sectionheader"){
    data.title = block.querySelector(".delf-section-title").innerHTML;
    data.points = block.querySelector(".delf-section-points").innerHTML;
  }
  if(type==="header"||type==="text"){
    data.content = block.querySelector("[contenteditable]").innerHTML;
  }
  if(type==="listening"||type==="reading"){
    const questions = Array.from(block.querySelectorAll(".delf-question-row")).map(row=>{
      const question = row.querySelector(".delf-question-text").innerHTML;
      const opts = Array.from(row.querySelectorAll(".delf-answer-opt"));
      const options = opts.map(o=>o.querySelector("span").innerHTML);
      let correct = 0;
      opts.forEach((o,i)=>{ if(o.querySelector("input").checked) correct = i; });
      return { question, options, correct };
    });
    data.questions = questions;
    if(type==="listening") data.audioSrc = block.dataset.audioSrc || "";
    if(type==="reading") data.passage = block.querySelector(".delf-passage-box").innerHTML;
  }
  if(type==="writing"){
    data.prompt = block.querySelector(".delf-prompt-box").innerHTML;
    data.wordCount = block.querySelector(".delf-wordcount").value;
    data.timeLimit = block.querySelector(".delf-timelimit").value;
  }
  if(type==="speaking"){
    data.prompt = block.querySelector(".delf-prompt-box").innerHTML;
    data.prepTime = block.querySelector(".delf-preptime").value;
    data.speakTime = block.querySelector(".delf-speaktime").value;
  }
  return data;
}

function serializeBlocks(){ return allBlocks().map(extractData); }

function rebuildFromBlocks(blocks){
  allBlocks().forEach(b=>b.remove());
  selectedBlock = null;
  if(!blocks || !blocks.length){ showEmpty(); return; }
  hideEmpty();
  blocks.forEach(data=>{
    const block = createBlock(data.type, data);
    if(!block) return;
    canvas.appendChild(block);
    if(data.type==="header"||data.type==="text"){
      const ce = block.querySelector("[contenteditable]");
      if(ce && data.content) ce.innerHTML = data.content;
    }
    if(data.type==="listening" && data.audioSrc){
      block.dataset.audioSrc = data.audioSrc;
    }
  });
}

async function save(){
  try{
    const blocks = serializeBlocks();
    const assignedLevels=Array.from(document.querySelectorAll("#assignLevelsGroup input:checked")).map(c=>c.value);
    const payload = {
createdByTeacher:(sessionStorage.getItem("impactAccessName")||sessionStorage.getItem("impactPersonnelName")||`"),
      title: titleInput.value.trim() || "Untitled Mock Exam",
      track: trackSelect.value,
      level: levelSelect.value,
      assignedLevels,
      assignedLearners:selectedLearners,
      blocks: blocks,
      version: 1,
      updatedAt: new Date().toISOString()
    };
    if(currentId){
      await db.collection("delfMocks").doc(currentId).set(payload, {merge:true});
    } else {
      const ref = await db.collection("delfMocks").add(payload);
      currentId = ref.id;
      history.replaceState(null,"","/teacher/delf-mock-builder.html?edit="+currentId);
    }
    tick();
    return currentId;
  }catch(err){
    console.error("Save failed:", err);
    alert("⚠️ Save failed: " + err.message);
    return null;
  }
}

let autoSaveTimer = null;
function scheduleAutoSave(){
  const hasTitle = titleInput.value.trim().length > 0;
  const hasBlocks = allBlocks().length > 0;
  if(!hasTitle || !hasBlocks) return;
  clearTimeout(autoSaveTimer);
  autoSaveTimer = setTimeout(async function(){ await save(); }, 1500);
}

  if(Array.isArray(d.blocks)) rebuildFromBlocks(d.blocks);
}

load();
