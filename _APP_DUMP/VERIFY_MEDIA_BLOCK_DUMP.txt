// ── CREATE BLOCK ──────────────────────────────────────────
function createBlock(type, data){
  const tmpl = TEMPLATES[type];
  if(!tmpl) return null;

  const block = document.createElement("div");
  block.className = "c-block";
  block.setAttribute("data-type", type);
  block.setAttribute("draggable", "true");

  const actions = document.createElement("div");
  actions.className = "block-actions";
  actions.innerHTML = `
    <button class="act-btn" data-act="up" title="Move up">↑</button>
    <button class="act-btn" data-act="down" title="Move down">↓</button>
    <div class="act-divider"></div>
    <button class="act-btn" data-act="dupe" title="Duplicate">⧉</button>
    <div class="act-divider"></div>
    <button class="act-btn danger" data-act="del" title="Delete">✕</button>
  `;

  const inner = document.createElement("div");
  inner.className = "c-block-inner";
  inner.innerHTML = tmpl.render(data);

  block.appendChild(actions);
  block.appendChild(inner);

  bindBlock(block);
  return block;
}

function bindBlock(block){
  // ── Align / resize toolbar ──
  const toolbar = block.querySelector(".block-toolbar");
  if(toolbar){
    toolbar.addEventListener("click", function(e){
      e.stopPropagation();
      const alignBtn = e.target.closest("[data-align]");
      if(alignBtn){
        pushUndo();
        block.classList.remove("align-left","align-center","align-right");
        block.classList.add("align-"+alignBtn.dataset.align);
        toolbar.querySelectorAll("[data-align]").forEach(b=>b.classList.remove("active"));
        alignBtn.classList.add("active");
      }
      const sizeBtn = e.target.closest("[data-size]");
      if(sizeBtn){
        pushUndo();
        block.classList.remove("bw-small","bw-medium","bw-large");
        block.classList.add("bw-"+sizeBtn.dataset.size);
        toolbar.querySelectorAll("[data-size]").forEach(b=>b.classList.remove("active"));
        sizeBtn.classList.add("active");
      }
    });
  }

  // ── Media File block: click to upload, preview immediately ──
  const mediaEmpty   = block.querySelector(".cb-media-empty");
  const mediaInput   = block.querySelector(".cb-media-input");
  const mediaPreview = block.querySelector(".cb-media-preview");
  const mediaReplace = block.querySelector(".cb-media-replace");

  function handleMediaFile(file){
    if(!file) return;
    const isVideo = file.type.startsWith("video/");
    const reader = new FileReader();
    reader.onload = function(ev){
      const src = ev.target.result;
      block.dataset.mediaSrc  = src;
      block.dataset.mediaKind = isVideo ? "video" : "image";

      mediaPreview.innerHTML = isVideo
        ? `<video src="${src}" controls></video>`
        : `<img src="${src}" alt="">`;
      mediaPreview.classList.add("show");
      if(mediaEmpty) mediaEmpty.style.display = "none";

      let replaceBtn = block.querySelector(".cb-media-replace");
      if(!replaceBtn){
        replaceBtn = document.createElement("button");
        replaceBtn.type = "button";
        replaceBtn.className = "cb-media-replace";
        replaceBtn.textContent = "Replace file";
        mediaPreview.after(replaceBtn);
        replaceBtn.addEventListener("click", function(e){
          e.stopPropagation();
          mediaInput.click();
        });
      }
    };
    if(isVideo){ reader.readAsDataURL(file); } else { reader.readAsDataURL(file); }
  }

  if(mediaEmpty && mediaInput){
    mediaEmpty.addEventListener("click", function(e){
      e.stopPropagation();
      mediaInput.click();
    });
    mediaInput.addEventListener("click", function(e){ e.stopPropagation(); });
    mediaInput.addEventListener("change", function(e){
      e.stopPropagation();
      handleMediaFile(mediaInput.files[0]);
    });
  }
  if(mediaReplace && mediaInput){
    mediaReplace.addEventListener("click", function(e){
      e.stopPropagation();
      mediaInput.click();
    });
  }
  block.addEventListener("click", function(e){

    selectBlock(block);

  const actionBar = block.querySelector(".block-actions");
  actionBar.addEventListener("click", function(e){

    if(!btn) return;
    const act = btn.dataset.act;

    if(act==="del"){
      pushUndo();
      block.remove();
      selectedBlock = null;
      showEmpty();
    }
    if(act==="up"){
      pushUndo();
      const prev = block.previousElementSibling;
      if(prev && prev.classList.contains("c-block")) canvas.insertBefore(block, prev);
    }
    if(act==="down"){
      pushUndo();
      const next = block.nextElementSibling;
      if(next && next.classList.contains("c-block")) canvas.insertBefore(next, block);
    }
    if(act==="dupe"){
      pushUndo();
      const type = block.dataset.type;
      const dupe = createBlock(type, extractData(block));
      block.after(dupe);
      selectBlock(dupe);
    }
  });

  // Add option button for multiple choice
  const mcAdd = block.querySelector(".mc-add");
  if(mcAdd){
    mcAdd.addEventListener("click", function(e){
      e.stopPropagation();
      pushUndo();
      const grid = block.querySelector(".mc-options");
      const opt = document.createElement("div");
      opt.className = "mc-option";
      opt.innerHTML = `<input type="radio" disabled><span class="mc-option-label" contenteditable="true">Option ${grid.children.length+1}</span>`;
      grid.appendChild(opt);
    });
  }

  // Wire up File Upload block: Browse Files button triggers hidden input,
  // selecting a file shows its name in the builder preview (preview only —
  // this is the builder canvas, not a live form; actual uploads happen
  // on the live-view.html page where learners fill out the form)
  const uploadBtn = block.querySelector(".impact-upload-btn");
  const hiddenFileInput = block.querySelector(".impact-hidden-upload");
  if(uploadBtn && hiddenFileInput){
    uploadBtn.addEventListener("click", function(e){
      e.stopPropagation();
      hiddenFileInput.click();
    });
    hiddenFileInput.addEventListener("click", function(e){ e.stopPropagation(); });
    hiddenFileInput.addEventListener("change", function(e){
      e.stopPropagation();
      const files = Array.from(hiddenFileInput.files);
      let nameEl = block.querySelector(".impact-upload-filename");
      if(!nameEl){
        nameEl = document.createElement("div");
        nameEl.className = "impact-upload-filename";
        hiddenFileInput.closest(".impact-drop-zone").appendChild(nameEl);
      }
      if(files.length){
        nameEl.textContent = files.length===1 ? "Selected: "+files[0].name : files.length+" files selected";
        nameEl.classList.add("show");
      }
    });
  }

  // Add image for picture choice
  const pcAdd = block.querySelector(".pc-add-img");
  if(pcAdd){
    pcAdd.addEventListener("click", function(e){
      e.stopPropagation();
      pcImageInput.click();
      pcImageInput.onchange = function(){
        const files = Array.from(pcImageInput.files);
        const grid = block.querySelector(".pc-grid");

        pcImageInput.value="";

  // Drag and drop reorder
  block.addEventListener("dragstart", function(e){
    dragSrc = block;
    e.dataTransfer.effectAllowed = "move";
    setTimeout(()=>block.style.opacity=".4", 0);
  });
  block.addEventListener("dragend", function(){
    block.style.opacity="";
    allBlocks().forEach(b=>b.classList.remove("drag-over"));
    dragSrc = null;
  });
  block.addEventListener("dragover", function(e){
    e.preventDefault();
    if(dragSrc && dragSrc !== block) block.classList.add("drag-over");
  });
  block.addEventListener("dragleave", function(){
    block.classList.remove("drag-over");
  });
  block.addEventListener("drop", function(e){
    e.preventDefault();
    block.classList.remove("drag-over");
    if(!dragSrc || dragSrc===block) return;
    pushUndo();
    const rect = block.getBoundingClientRect();
    if(e.clientY < rect.top + rect.height/2) canvas.insertBefore(dragSrc, block);
    else block.after(dragSrc);
  });
}

// ── SERIALISE / DESERIALISE ───────────────────────────────
function extractData(block){
  const type = block.dataset.type;
  const data = { type };

  if(type==="header"||type==="subheader"||type==="text"){
    data.content = block.querySelector("[contenteditable]").innerHTML;
  }
  if(type==="pagebreak"){
    const wrap = block.querySelector(".page-break-wrap");
    data.pageNum = wrap ? parseInt(wrap.dataset.page)||2 : 2;
  }
  if(type==="shorttext"||type==="longtext"||type==="date"||type==="number"||type==="website"||type==="fileupload"||type==="signature"||type==="mediaanswer"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
  }
  if(type==="multiplechoice"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.options = Array.from(block.querySelectorAll(".mc-option-label")).map(el=>el.innerHTML);
  }
  if(type==="picturechoice"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.images = Array.from(block.querySelectorAll(".pc-img-placeholder img")).map(img=>img.src);
  }
  if(type==="dropdown"){
    const labelEl = block.querySelector(".q-label-text");
    const hintEl = block.querySelector(".q-hint");
    data.label = labelEl ? labelEl.innerHTML : "";
    data.hint = hintEl ? hintEl.innerHTML : "";
    data.options = Array.from(block.querySelectorAll("select option")).slice(1).map(o=>o.textContent);
  }
    // Persist media file data URL + block styling so it survives save/reload
  if(type==="mediafile"){
    data.src  = block.dataset.mediaSrc  || "";
    data.kind = block.dataset.mediaKind || "";
  }
  const alignClass = ["align-left","align-center","align-right"].find(c=>block.classList.contains(c));
  const sizeClass   = ["bw-small","bw-medium","bw-large"].find(c=>block.classList.contains(c));
  data.align = alignClass ? alignClass.replace("align-","") : "center";
  data.size  = sizeClass  ? sizeClass.replace("bw-","")     : "large";
  return data;
}

function serializeBlocks(){
  return allBlocks().map(extractData);
}

function rebuildFromBlocks(blocks){
  allBlocks().forEach(b=>b.remove());
  selectedBlock = null;
  pageCount = 1;
  if(!blocks || !blocks.length){ showEmpty(); return; }
  hideEmpty();
  blocks.forEach(data=>{
    const block = createBlock(data.type, data);
    if(!block) return;
    canvas.appendChild(block);
    // Restore editable content
    if(data.content){
      const ce = block.querySelector("[contenteditable]");
      if(ce) ce.innerHTML = data.content;
    }
    if(data.label){
      const lel = block.querySelector(".q-label-text");
      if(lel) lel.innerHTML = data.label;
    }
    if(data.hint){
      const hel = block.querySelector(".q-hint");
      if(hel) hel.innerHTML = data.hint;
    }
  });
}

async function save(){
  const blocks = serializeBlocks();
  const payload = {
    title: titleInput.value.trim() || "Untitled Lesson",
    level: levelSelect.value,
    blocks: blocks,
    version: 2,
    updatedAt: new Date().toISOString()
  };
  if(currentId){
    await db.collection("lessonPlans").doc(currentId).set(payload, {merge:true});
  } else {
    const ref = await db.collection("lessonPlans").add(payload);
    currentId = ref.id;
    history.replaceState(null,"","/teacher/lesson-builder.html?edit="+currentId);
  }
  tick();
  return currentId;
}

// Auto-save fires when title or level changes, but ONLY when:
// 1. A level has been selected (not the blank default)
// 2. A title has been entered
// 3. At least one block exists on the canvas
let autoSaveTimer = null;
function scheduleAutoSave(){
  const hasTitle = titleInput.value.trim().length > 0;
  const hasBlocks = allBlocks().length > 0;
  const hasLevel = levelSelect.value.length > 0;
  if(!hasTitle || !hasBlocks || !hasLevel) return;
  clearTimeout(autoSaveTimer);
  autoSaveTimer = setTimeout(async function(){
    await save();
  }, 1200);
}
titleInput.addEventListener("input", scheduleAutoSave);
levelSelect.addEventListener("change", scheduleAutoSave);

// Also trigger auto-save when blocks are added/removed
const canvasObserver = new MutationObserver(function(){
  scheduleAutoSave();
});
canvasObserver.observe(canvas, { childList: true, subtree: false });

// ── LOAD ─────────────────────────────────────────────────
async function load(){
  if(!currentId) return;
  const doc = await db.collection("lessonPlans").doc(currentId).get();
  if(!doc.exists) return;
  const d = doc.data();
  titleInput.value = d.title || d.name || "";
  if(d.level) levelSelect.value = d.level;

  // JSON (v2) plan
  if(d.version===2 && Array.isArray(d.blocks)){
    rebuildFromBlocks(d.blocks);
    return;
  }

  // Legacy HTML plan — display read-only notice
  if(d.content){
    hideEmpty();
    const notice = document.createElement("div");
    notice.className = "legacy-notice";
    notice.innerHTML = "⚠️ This lesson plan was created with the previous builder and is shown in read-only mode. Use <strong>New +</strong> to start a fresh plan with the new components.";
    canvas.insertBefore(notice, canvas.firstChild);
    const wrapper = document.createElement("div");
    wrapper.style.cssText = "padding:12px;border-radius:10px;border:1.5px solid #e0e3e8;background:#fafbfc;color:#374151;font-size:.88rem;line-height:1.7;pointer-events:none";
    wrapper.innerHTML = d.content;
    canvas.appendChild(wrapper);
  }
}

document.addEventListener("DOMContentLoaded", async function(){
  const db = firebase.firestore();
  const id = new URLSearchParams(location.search).get("id");
  const slideCard = document.getElementById("slideCard");
  const titleBar = document.getElementById("titleBar");
  const slideCounter = document.getElementById("slideCounter");
  const progressFill = document.getElementById("progressFill");
  const prevBtn = document.getElementById("prevBtn");
  const nextBtn = document.getElementById("nextBtn");

  if(!id){
    slideCard.innerHTML = '<div class="centered" style="color:#374151">No lesson ID provided.</div>';
    return;
  }

  // Load from Firestore directly
  let plan;
  try{
    const snap = await db.collection("lessonPlans").doc(id).get();
    if(!snap.exists){ slideCard.innerHTML='<div class="centered" style="color:#374151">Lesson plan not found.</div>'; return; }
    plan = { id: snap.id, ...snap.data() };
  }catch(err){
    slideCard.innerHTML = '<div class="centered" style="color:#374151">Could not load lesson. Please try again.</div>';
    return;
  }

  titleBar.textContent = plan.title || plan.name || "Lesson Plan";
  document.title = (plan.title || "Lesson Plan") + " | Preview";

  // ── SPLIT BLOCKS INTO PAGES ───────────────────────────────
  // Each page break creates a new slide. Blocks before first break = slide 1.
  let slides = [[]];
  let blocks = [];

  if(plan.version === 2 && Array.isArray(plan.blocks)){
    blocks = plan.blocks;
  } else if(plan.content){
    // Legacy HTML — show as single slide
    slides = [[{ type:"__html__", content: plan.content }]];
  }

  if(blocks.length){
    blocks.forEach(b=>{
      if(b.type === "pagebreak"){
        slides.push([]);
      } else {
        slides[slides.length - 1].push(b);
      }
    });
    // Remove empty trailing slide
    if(slides[slides.length-1].length === 0) slides.pop();
  }

  // Add a final "submit" slide
  const totalContent = slides.length;
  slides.push([{ type:"__final__" }]);

  let current = 0;

  function renderBlock(b){
    if(b.type === "__html__"){
      return `<div style="color:#374151;font-size:.9rem;line-height:1.7">${b.content}</div>`;
    }
    if(b.type === "__final__"){
      return `<div class="final-slide"><h2>All done! 🎉</h2><p>You have completed this lesson plan.</p><button class="final-submit" onclick="window.close()">Close Preview</button></div>`;
    }
    if(b.type==="mediafile"){
      if(!b.src) return "";
      const mediaTag = b.kind==="video"
        ? `<video src="${b.src}" controls style="width:100%;border-radius:12px;max-height:460px"></video>`
        : `<img src="${b.src}" alt="" style="width:100%;border-radius:12px;max-height:460px;object-fit:contain;background:#0a0c14">`;
      return `<div>${mediaTag}</div>`;
    }
    if(b.type==="header") return `<div class="sv-header">${b.content||""}</div>`;
    if(b.type==="subheader") return `<div class="sv-subheader">${b.content||""}</div>`;
    if(b.type==="text") return `<div class="sv-text">${b.content||""}</div>`;

    const label = b.label || "";
    const hint = b.hint || "";
    const qHtml = `<div class="q-row"><span class="q-badge">?</span><div class="q-text">${label}</div></div>${hint?`<div class="q-hint">${hint}</div>`:""}`;

    if(b.type==="shorttext") return `${qHtml}<input class="sv-input" type="text" placeholder="Short answer...">`;
    if(b.type==="longtext") return `${qHtml}<textarea class="sv-textarea" placeholder="Long answer..."></textarea>`;
    if(b.type==="date") return `${qHtml}<input class="sv-date" type="date">`;
    if(b.type==="number") return `${qHtml}<input class="sv-number" type="number" placeholder="0">`;
    if(b.type==="website") return `${qHtml}<input class="sv-url" type="url" placeholder="https://">`;
    if(b.type==="fileupload"){
      const fid = "file_" + Math.random().toString(36).slice(2,9);
      return `${qHtml}<div class="sv-file-zone-wrap">
        <input type="file" id="${fid}" class="sv-file-input" onchange="
          const f=this.files[0];
          const lbl=this.nextElementSibling;
          if(f){
            lbl.classList.add('has-file');
            lbl.innerHTML='📎 <span class=sv-file-name>'+f.name+'</span><br><span style=font-size:.72rem;opacity:.6>Click to change file</span>';
          }
        ">
        <label for="${fid}" class="sv-file-label">📎 Click to upload a file</label>
      </div>`;
    }
    if(b.type==="signature") return `${qHtml}<div class="sv-sig">Sign here</div>`;

    if(b.type==="multiplechoice"){
      const opts = (b.options||[]).map(o=>`<label class="mc-opt"><input type="radio" name="mc_${Math.random()}"> ${o}</label>`).join("");
      return `${qHtml}<div class="mc-grid">${opts}</div>`;
    }
    if(b.type==="picturechoice"){
      const imgs = (b.images||[]);
      const items = imgs.map((src,i)=>`<div class="pc-item"><div>${src?`<img src="${src}" alt="Option ${i+1}">`:'<div class="pc-item-ph">📷</div>'}</div><div class="pc-label"><input type="radio"> Option ${i+1}</div></div>`).join("");
      return `${qHtml}<div class="pc-grid">${items}</div>`;
    }
    if(b.type==="dropdown"){
      const opts = (b.options||[]).map(o=>`<option>${o}</option>`).join("");
      return `${qHtml}<select class="sv-select"><option value="">Choose an option...</option>${opts}</select>`;
    }
    if(b.type==="mediaanswer"){
      return `${qHtml}<div class="ma-btns">
        <button class="ma-btn"><svg viewBox="0 0 24 24"><path d="M15 10l4.553-2.276A1 1 0 0 1 21 8.723v6.554a1 1 0 0 1-1.447.894L15 14v-4z"/><rect x="2" y="6" width="13" height="12" rx="2"/></svg>VIDEO</button>
        <button class="ma-btn"><svg viewBox="0 0 24 24"><path d="M12 1a3 3 0 0 0-3 3v8a3 3 0 0 0 6 0V4a3 3 0 0 0-3-3z"/><path d="M19 10v2a7 7 0 0 1-14 0v-2M12 19v4M8 23h8"/></svg>AUDIO</button>
        <button class="ma-btn"><svg viewBox="0 0 24 24"><rect x="2" y="3" width="20" height="14" rx="2"/><path d="M8 21h8M12 17v4"/></svg>SCREEN</button>
        <button class="ma-btn"><svg viewBox="0 0 24 24"><path d="M11 4H4a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-7"/><path d="M18.5 2.5a2.121 2.121 0 0 1 3 3L12 15l-4 1 1-4 9.5-9.5z"/></svg>TEXT</button>
      </div>`;
    }
    return "";
  }

  function renderSlide(index){
    const slide = slides[index];
    const isFinal = slide.length === 1 && slide[0].type === "__final__";

    slideCard.style.animation = "none";
    slideCard.offsetHeight; // reflow
    slideCard.style.animation = "";

    slideCard.innerHTML = slide.map(renderBlock).filter(Boolean).join('<div style="height:12px"></div>');

    const total = slides.length;
    const pct = ((index) / (total - 1)) * 100;
    progressFill.style.width = pct + "%";

    if(isFinal){
      slideCounter.textContent = "Complete";
      nextBtn.style.display = "none";
    } else {
      slideCounter.textContent = `${index + 1} / ${totalContent}`;
      nextBtn.style.display = "";
      nextBtn.textContent = index === totalContent - 1 ? "Submit" : "Continue →";
    }

    prevBtn.disabled = index === 0;
    slideCard.scrollTop = 0;
  }

  prevBtn.addEventListener("click", function(){
    if(current > 0){ current--; renderSlide(current); }
  });

  nextBtn.addEventListener("click", function(){
    if(current < slides.length - 1){ current++; renderSlide(current); }
  });

  // Keyboard navigation
  document.addEventListener("keydown", function(e){
    if(e.key==="ArrowRight"||e.key==="Enter") nextBtn.click();
    if(e.key==="ArrowLeft") prevBtn.click();
  });

  renderSlide(0);
});

document.addEventListener("DOMContentLoaded", async function(){
  const db=firebase.firestore();
  const id=new URLSearchParams(location.search).get("id");
  const slideCard=document.getElementById("slideCard");
  const titleBar=document.getElementById("titleBar");
  const slideCounter=document.getElementById("slideCounter");
  const progressFill=document.getElementById("progressFill");
  const prevBtn=document.getElementById("prevBtn");
  const nextBtn=document.getElementById("nextBtn");
  if(!id){slideCard.innerHTML='<div style="text-align:center;padding:48px;color:#374151;font-weight:700">No activity ID provided.</div>';return;}
  let plan;
  try{const snap=await db.collection("activityPlans").doc(id).get();if(!snap.exists){slideCard.innerHTML='<div style="text-align:center;padding:48px;color:#374151;font-weight:700">Activity not found.</div>';return;}plan={id:snap.id,...snap.data()};}
  catch(err){slideCard.innerHTML='<div style="text-align:center;padding:48px;color:#374151;font-weight:700">Could not load activity.</div>';return;}
  titleBar.textContent=plan.title||"Activity";
  document.title=(plan.title||"Activity")+" | Preview";
  let slides=[[]];let blocks=[];
  if(plan.version===2&&Array.isArray(plan.blocks)){blocks=plan.blocks;}
  else if(plan.content){slides=[[{type:"__html__",content:plan.content}]];}
  if(blocks.length){blocks.forEach(b=>{if(b.type==="pagebreak"){slides.push([]);}else{slides[slides.length-1].push(b);}});if(slides[slides.length-1].length===0)slides.pop();}
  const totalContent=slides.length;
  slides.push([{type:"__final__"}]);
  let current=0;
  function renderBlock(b){
    if(b.type==="__html__")return `<div style="color:#374151;font-size:.9rem;line-height:1.7">${b.content}</div>`;
    if(b.type==="__final__")return `<div class="final-slide"><h2>Activity complete! 🎉</h2><p>You have finished this activity.</p><button class="final-submit" onclick="window.close()">Close Preview</button></div>`;
    if(b.type==="header")return `<div class="sv-header">${b.content||""}</div>`;
    if(b.type==="subheader")return `<div class="sv-subheader">${b.content||""}</div>`;
    if(b.type==="text")return `<div class="sv-text">${b.content||""}</div>`;
    const label=b.label||"";const hint=b.hint||"";
    const qHtml=`<div class="q-row"><span class="q-badge">?</span><div class="q-text">${label}</div></div>${hint?`<div class="q-hint">${hint}</div>`:""}`;
    if(b.type==="shorttext")return `${qHtml}<input class="sv-input" type="text" placeholder="Short answer...">`;
    if(b.type==="longtext")return `${qHtml}<textarea class="sv-textarea" placeholder="Long answer..."></textarea>`;
    if(b.type==="date")return `${qHtml}<input class="sv-date" type="date">`;
    if(b.type==="number")return `${qHtml}<input class="sv-number" type="number" placeholder="0">`;
    if(b.type==="website")return `${qHtml}<input class="sv-url" type="url" placeholder="https://">`;
    if(b.type==="fileupload")return `${qHtml}<div class="sv-file-zone">📎 Click to upload</div>`;
    if(b.type==="signature")return `${qHtml}<div class="sv-sig">Sign here</div>`;
    if(b.type==="multiplechoice"){const opts=(b.options||[]).map(o=>`<label class="mc-opt"><input type="radio" name="mc_${Math.random()}"> ${o}</label>`).join("");return `${qHtml}<div class="mc-grid">${opts}</div>`;}
    if(b.type==="picturechoice"){const imgs=(b.images||[]);const items=imgs.map((src,i)=>`<div class="pc-item"><div>${src?`<img src="${src}" alt="Option ${i+1}">`:'<div class="pc-item-ph">📷</div>'}</div><div class="pc-label"><input type="radio"> Option ${i+1}</div></div>`).join("");return `${qHtml}<div class="pc-grid">${items}</div>`;}
    if(b.type==="dropdown"){const opts=(b.options||[]).map(o=>`<option>${o}</option>`).join("");return `${qHtml}<select class="sv-select"><option value="">Choose an option...</option>${opts}</select>`;}

    return "";
  }
  function renderSlide(index){
    const slide=slides[index];const isFinal=slide.length===1&&slide[0].type==="__final__";
    slideCard.style.animation="none";slideCard.offsetHeight;slideCard.style.animation="";
    slideCard.innerHTML=slide.map(renderBlock).filter(Boolean).join('<div style="height:12px"></div>');
    const pct=(index/(slides.length-1))*100;progressFill.style.width=pct+"%";
    if(isFinal){slideCounter.textContent="Complete";nextBtn.style.display="none";}

    prevBtn.disabled=index===0;slideCard.scrollTop=0;
