async function deleteRec(btn){
  if(!confirm("Delete this recording permanently?")) return;
  const db = firebase.firestore();
  try{
    await db.collection("recordings").doc(btn.dataset.id).delete();
    btn.closest(".rec-card").remove();
    const grid = document.getElementById("recGrid");
    if(!grid.querySelector(".rec-card")){
      grid.innerHTML = `<div class="empty-state"><h3>No recordings yet</h3></div>`;
    }
  }catch(e){
    alert("Could not delete: " + e.message);
  }
}

function docRef(){
  return db.collection("academicCalendars").doc(String(currentYear));
}

function normalizeCalendarData(){
  if(!data.months) data.months = {};

  monthNames.forEach(month => {
    if(!Array.isArray(data.months[month])) data.months[month] = [];

    data.months[month] = data.months[month].map(item => {
      if(Array.isArray(item) && item.length >= 4){
        return [item[0], item[2], null];
      }

      if(Array.isArray(item) && item.length === 3){
        if(typeof item[2] === "boolean" || item[2] === null){
          return [item[0], item[1], item[2]];
        }

        return [item[0], item[2], null];
      }

      if(Array.isArray(item) && item.length === 2){
        return [item[0], item[1], null];
      }

      return ["", String(item || ""), null];
    });
  });
}

async function load(){
  yearLabel.textContent = currentYear;
  statusEl.textContent = "Loading calendar...";

  const fallback = () => {
    data = JSON.parse(JSON.stringify({ months: starter, notes:"" }));
    normalizeCalendarData();
    notesEl.value = data.notes || "";
    render();
    statusEl.textContent = "Calendar loaded offline. Firestore will sync when available.";
  };

  try{
    const timeout = new Promise((_, reject) =>
      setTimeout(() => reject(new Error("Firestore timeout")), 4500)
    );

    const snap = await Promise.race([docRef().get(), timeout]);

    if(snap.exists){
      data = snap.data();
    }else{
      data = JSON.parse(JSON.stringify({ months: starter, notes:"" }));
      await docRef().set(data,{merge:true});
    }

    normalizeCalendarData();
    notesEl.value = data.notes || "";
    render();

    statusEl.textContent = "Saved in Firestore";
  }catch(err){
    console.warn("Firestore load failed:", err);
    fallback();
  }
}

async function save(){
  normalizeCalendarData();
  data.notes = notesEl.value;

  try{
    await docRef().set(data,{merge:true});
    statusEl.textContent = "Saved ✓";
    setTimeout(()=>statusEl.textContent="Saved in Firestore",1000);
  }catch(err){
    console.warn("Firestore save failed:", err);
    localStorage.setItem("impactAcademicCalendar_" + currentYear, JSON.stringify(data));
    statusEl.textContent = "Saved locally. Firestore not reachable.";
  }
}

function makeStatus(month,index,item,row){
  const wrap = document.createElement("div");
  wrap.className = "status-actions";

  const yes = document.createElement("button");
  yes.className = "status-btn ok" + (item[2] === true ? " active" : "");
  yes.type = "button";

  const no = document.createElement("button");
  no.className = "status-btn no" + (item[2] === false ? " active" : "");
  no.type = "button";

  yes.onclick = async () => {
    data.months[month][index][2] = true;
    row.classList.add("done");
    yes.classList.add("active");
    no.classList.remove("active");
    await save();
  };

  no.onclick = async () => {
    data.months[month][index][2] = false;
    row.classList.remove("done");
    no.classList.add("active");
    yes.classList.remove("active");
    await save();
  };

  wrap.appendChild(yes);
  wrap.appendChild(no);

  return wrap;
}

function render(){
  normalizeCalendarData();
  monthsEl.innerHTML = "";

  monthNames.forEach(month => {
    const card = document.createElement("article");
    card.className = "card";

    const head = document.createElement("div");
    head.className = "card-head";
    head.innerHTML = '<span class="head-icon">▦</span>' + month;

    const list = document.createElement("div");
    list.className = "items";

    data.months[month].forEach((item,index) => {
      const row = document.createElement("div");
      row.className = "item";
      row.classList.toggle("done", item[2] === true);

      const date = document.createElement("div");
      date.className = "date";
      date.textContent = item[0] || "";

      const text = document.createElement("div");
      text.className = "event-text";
      text.textContent = item[1] || "";

      const status = makeStatus(month,index,item,row);

      const del = document.createElement("button");
      del.className = "delete";
      del.type = "button";
      del.textContent = "×";
      del.onclick = async () => {
        data.months[month].splice(index,1);
        render();
        await save();
      };

      row.appendChild(date);
      row.appendChild(text);
      row.appendChild(status);
      row.appendChild(del);
      list.appendChild(row);
    });

    const add = document.createElement("div");
    add.className = "add-line";
    add.innerHTML = '<input placeholder="Date"><input placeholder="Event"><button type="button">+</button>';

    add.querySelector("button").onclick = async () => {
      const inputs = add.querySelectorAll("input");
      const date = inputs[0].value.trim();
      const event = inputs[1].value.trim();

      if(!date || !event) return;

      data.months[month].push([date,event,null]);

      inputs[0].value = "";
      inputs[1].value = "";

      render();
      await save();
    };

    card.appendChild(head);
    card.appendChild(list);
    card.appendChild(add);
    monthsEl.appendChild(card);
  });
}
