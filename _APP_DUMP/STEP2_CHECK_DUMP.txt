      if(delBtn){
        e.stopPropagation();
        if(delBtn.dataset.pub === "1"){
          const auth = await window.ImpactPublish.requirePasskey("Delete published activity", true);
          if(!auth) return;
          await window.ImpactPublish.logAction({
            action:"delete", itemType:"activity", itemId:delBtn.dataset.id,
            itemTitle:delBtn.dataset.title, performedBy:auth.name, performedRole:auth.role
          });
        }else{
          const anchorRow = delBtn.closest(".plan-row");
          const ok = await showDeletePopover(anchorRow, `Delete &ldquo;${delBtn.dataset.title||"this activity"}&rdquo;?`);
          if(!ok) return;
        }
        await db.collection("activityPlans").doc(delBtn.dataset.id).delete();
        allPlans = allPlans.filter(p=>p.id!==delBtn.dataset.id);
        const level = delBtn.closest(".folder-card").dataset.level;
        if(!allPlans.filter(p=>(p.level||"Uncategorised")===level).length) openFolder=null;
        render(); return;
      }
      const viewBtn = e.target.closest(".view-btn");
      if(viewBtn){ e.stopPropagation(); location.href="/teacher/activity-view.html?id="+encodeURIComponent(viewBtn.dataset.id); return; }
      const editBtn = e.target.closest(".edit-btn");
      if(editBtn){
        e.stopPropagation();
        if(editBtn.dataset.pub === "1"){
          const auth = await window.ImpactPublish.requirePasskey("Edit published activity", false);
          if(!auth) return;
          await window.ImpactPublish.logAction({
            action:"edit", itemType:"activity", itemId:editBtn.dataset.id,
            itemTitle:editBtn.dataset.title, performedBy:auth.name, performedRole:auth.role
          });
        }
        location.href="/teacher/activity-builder.html?edit="+encodeURIComponent(editBtn.dataset.id);
        return;
      }
      const card = e.target.closest(".folder-card");
      if(card){
        const level = card.dataset.level;
        if(openFolder===level){ openFolder=null; card.classList.remove("open"); card.querySelector(".folder-icon").textContent="📁"; }
        else {
          document.querySelectorAll(".folder-card.open").forEach(c=>{ c.classList.remove("open"); c.querySelector(".folder-icon").textContent="📁"; });
          openFolder=level; card.classList.add("open"); card.querySelector(".folder-icon").textContent="📂";
        }
      }
    });
  }

  await loadPlans();

// ---- SHARE DIALOG ----
let shareId = null;
let shareLearners = [];
let allLearnersCache = [];

async function loadLearnersOnce(){
  if(allLearnersCache.length) return;
  try{
    const snap = await db.collection("impactPasskeys").where("role","==","Learner").get();
    allLearnersCache = snap.docs.map(function(d){
      return { name: d.data().name || "Unnamed", level: d.data().level || "" };
    });
  }catch(e){ console.warn("learners load:", e); }
}

function renderShareChips(){
  const wrap = document.getElementById("shareSelected");
  if(!wrap) return;
  wrap.innerHTML = shareLearners.map(function(n){
    return '<span class="assigned-learner-chip">' + n + '<button type="button" data-rm="' + n + '">x</button></span>';
  }).join("");
}

async function openShareDialog(id, title){
  shareId = id;
  await loadLearnersOnce();
  const doc = await db.collection("activityPlans").doc(id).get();
  const d = doc.exists ? doc.data() : {};
  shareLearners = Array.isArray(d.assignedLearners) ? d.assignedLearners.slice() : [];

  document.getElementById("shareTitle").textContent = "Share: " + (title || "Activity");
  document.querySelectorAll("#shareLevels input").forEach(function(cb){
    cb.checked = Array.isArray(d.assignedLevels) && d.assignedLevels.indexOf(cb.value) >= 0;
    cb.closest(".assign-level-chip").classList.toggle("checked", cb.checked);
  });
  document.getElementById("shareSearch").value = "";
  document.getElementById("sharePicker").classList.remove("show");
  renderShareChips();
  document.getElementById("shareOverlay").classList.add("show");
}

document.getElementById("shareLevels").addEventListener("change", function(e){
  const chip = e.target.closest(".assign-level-chip");
  if(chip) chip.classList.toggle("checked", e.target.checked);
});

document.getElementById("shareSearch").addEventListener("input", function(){
  const q = this.value.trim().toLowerCase();
  const list = document.getElementById("sharePicker");
  if(!q){ list.classList.remove("show"); list.innerHTML=""; return; }
  const matches = allLearnersCache.filter(function(l){
    return l.name.toLowerCase().indexOf(q) >= 0 && shareLearners.indexOf(l.name) < 0;
  });
  if(!matches.length){ list.classList.remove("show"); list.innerHTML=""; return; }
  list.innerHTML = matches.map(function(l){
    return '<div class="learner-picker-row" data-pick="' + l.name + '">' + l.name + ' <span style="opacity:.5">(' + (l.level||"no level") + ')</span></div>';
  }).join("");
  list.classList.add("show");
});

document.getElementById("sharePicker").addEventListener("click", function(e){
  const row = e.target.closest("[data-pick]");
  if(!row) return;
  if(shareLearners.indexOf(row.dataset.pick) < 0) shareLearners.push(row.dataset.pick);
  renderShareChips();
  document.getElementById("shareSearch").value = "";
  this.classList.remove("show");
  this.innerHTML = "";
});

document.getElementById("shareSelected").addEventListener("click", function(e){
  const btn = e.target.closest("[data-rm]");
  if(!btn) return;
  shareLearners = shareLearners.filter(function(n){ return n !== btn.dataset.rm; });
  renderShareChips();
});

document.getElementById("shareCancel").addEventListener("click", function(){
  document.getElementById("shareOverlay").classList.remove("show");
});

document.getElementById("shareSave").addEventListener("click", async function(){
  if(!shareId) return;
  const levels = Array.from(document.querySelectorAll("#shareLevels input:checked")).map(function(c){ return c.value; });
  const btn = this;
  btn.disabled = true; btn.textContent = "Saving...";
  try{
    await db.collection("activityPlans").doc(shareId).set({
      assignedLevels: levels,
      assignedLearners: shareLearners,
      sharedAt: new Date().toISOString(),
      sharedBy: sessionStorage.getItem("impactAccessName") || sessionStorage.getItem("impactPersonnelName") || ""
    }, { merge:true });
    document.getElementById("shareOverlay").classList.remove("show");
    await load();
  }catch(e){
    alert("Could not save sharing: " + e.message);
  }finally{
    btn.disabled = false; btn.textContent = "Save Sharing";
  }
});
});
