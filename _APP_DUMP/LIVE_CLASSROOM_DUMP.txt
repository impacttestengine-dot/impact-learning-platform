function startRecording(){
  recordChunks=[];
  const mimeType=
    MediaRecorder.isTypeSupported("video/webm;codecs=vp9")?"video/webm;codecs=vp9":
    MediaRecorder.isTypeSupported("video/webm")?"video/webm":"video/mp4";

  const canvas=document.createElement("canvas");
  canvas.width=1280;canvas.height=720;
  const ctx=canvas.getContext("2d");
  const cols=Math.min(Object.keys(peers).length+1,3)||1;
  const rows=Math.ceil((Object.keys(peers).length+1)/cols);
  const tw=canvas.width/cols;const th=canvas.height/rows;

  const drawInterval=setInterval(()=>{
    ctx.fillStyle="#0a0c14";ctx.fillRect(0,0,canvas.width,canvas.height);
    Array.from(videoGrid.querySelectorAll("video")).filter(v=>v.srcObject).forEach((v,i)=>{
      const col=i%cols;const row=Math.floor(i/cols);
      try{ctx.drawImage(v,col*tw+2,row*th+2,tw-4,th-4);}catch(e){}
    });
  },1000/24);

  const cs=canvas.captureStream(24);

  if(localStream){
    try{
      const ac=new AudioContext();
      const dest=ac.createMediaStreamDestination();
      ac.createMediaStreamSource(localStream).connect(dest);
      Object.values(remoteStreams).forEach(rs=>{
        try{ac.createMediaStreamSource(rs).connect(dest);}catch(e){}
      });
      if(dest.stream.getAudioTracks().length) cs.addTrack(dest.stream.getAudioTracks()[0]);
    }catch(e){}
  }

  try{ mediaRecorder=new MediaRecorder(cs,{mimeType,videoBitsPerSecond:2500000}); }
  catch(e){ mediaRecorder=new MediaRecorder(cs); }

  mediaRecorder.ondataavailable=e=>{if(e.data.size>0)recordChunks.push(e.data);};
  mediaRecorder.onstop=async()=>{
    clearInterval(drawInterval);
    await uploadRecording(mimeType);
  };
  mediaRecorder.start(1000);
  recording=true;
  document.getElementById("recBtn").classList.add("recording");
  document.getElementById("recLabel").textContent="Stop Rec";
  toast("🔴 Recording started");
}

function stopRecording(){
  if(mediaRecorder&&recording){
    mediaRecorder.stop();recording=false;
    document.getElementById("recBtn").classList.remove("recording");
    document.getElementById("recLabel").textContent="Uploading...";
    document.getElementById("recBtn").disabled=true;
    toast("Recording stopped — uploading...",5000);
  }
}

async function uploadRecording(mimeType){
  try{
    const ext=mimeType.includes("mp4")?"mp4":"webm";
    const blob=new Blob(recordChunks,{type:mimeType});
    const path=`recordings/${MEETING_ID}/recording_${Date.now()}.${ext}`;
    const ref=storage.ref(path);
    const snap=await ref.put(blob);
    const url=await snap.ref.getDownloadURL();
    const dur=Math.round((Date.now()-startTime)/1000);
    const recId="rec_"+Date.now();
    await db.collection("recordings").doc(recId).set({
      recordingId:recId,meetingId:MEETING_ID,meetingTitle,
      storagePath:path,downloadURL:url,duration:dur,mimeType,
      createdAt:new Date().toISOString(),recordedBy:MY_NAME
    });
    // Update meeting with recording ref
    await db.collection("meetings").doc(MEETING_ID).update({
      hasRecording:true,lastRecordingId:recId
    }).catch(()=>{});
    document.getElementById("recLabel").textContent="Record";
    document.getElementById("recBtn").disabled=false;
    toast("✅ Recording saved!",4000);
  }catch(e){
    console.error("Upload failed:",e);
    document.getElementById("recLabel").textContent="Record";
    document.getElementById("recBtn").disabled=false;
    toast("⚠️ Upload failed: "+e.message,5000);
  }
}

    const snap = await db.collection("recordings").orderBy("createdAt","desc").get();

    if(snap.empty){
      grid.innerHTML = `<div class="empty-state">
        <h3>No recordings yet</h3>
        <p>Start a Live Classroom session and record it.<br>Recordings appear here automatically.</p>
      </div>`;
      return;
    }

    grid.innerHTML = "";

    snap.docs.forEach(doc=>{
      const r    = doc.data();
      const date = r.createdAt ? new Date(r.createdAt).toLocaleString() : "Unknown date";
      const card = document.createElement("div");
      card.className = "rec-card";
      card.innerHTML = `
        <div class="rec-card-title" title="${r.meetingTitle||r.meetingId||"Recording"}">${r.meetingTitle||"Untitled Meeting"}</div>
        <div class="rec-card-meta">
          📅 ${date}<br>
          🆔 ${r.meetingId||""}<br>
          👤 Recorded by ${r.recordedBy||"Unknown"}
        </div>
        <div class="rec-actions">
          <a class="rec-btn" href="${r.downloadURL}" target="_blank" rel="noopener">
            <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><polygon points="5 3 19 12 5 21 5 3"/></svg>
            Play
          </a>
          <a class="rec-btn" href="${r.downloadURL}" download target="_blank" rel="noopener">
            <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"/><polyline points="7 10 12 15 17 10"/><line x1="12" y1="15" x2="12" y2="3"/></svg>
            Download
          </a>
          <button class="rec-btn" data-id="${doc.id}" data-path="${r.storagePath||""}" onclick="deleteRec(this)" style="border-color:rgba(255,100,100,.25);color:#feb2b2">
            <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><polyline points="3 6 5 6 21 6"/><path d="M19 6l-1 14a2 2 0 0 1-2 2H8a2 2 0 0 1-2-2L5 6"/><path d="M10 11v6M14 11v6"/></svg>
            Delete
          </button>
        </div>`;
      grid.appendChild(card);
    });
  }catch(e){
    grid.innerHTML = `<div class="empty-state"><h3>Could not load recordings</h3><p>${e.message}</p></div>`;
  }
});

async function deleteRec(btn){
  if(!confirm("Delete this recording permanently?")) return;
  const db = firebase.firestore();
  try{
    await db.collection("recordings").doc(btn.dataset.id).delete();
    btn.closest(".rec-card").remove();
    const grid = document.getElementById("recGrid");
    if(!grid.querySelector(".rec-card")){
      grid.innerHTML = `<div class="empty-state"><h3>No recordings yet</h3></div>`;
    }
  }catch(e){
    alert("Could not delete: " + e.message);
  }
}
