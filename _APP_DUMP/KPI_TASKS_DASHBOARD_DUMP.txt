document.addEventListener("DOMContentLoaded", async()=>{
  const db=firebase.firestore();

  const form=document.getElementById("taskForm");
  const taskTitle=document.getElementById("taskTitle");
  const assignedTo=document.getElementById("assignedTo");
  const dueDate=document.getElementById("dueDate");
  const list=document.getElementById("taskList");

  let tasks=[];

  function fmtDate(v){
    if(!v) return "";
    const d=new Date(v);
    if(isNaN(d)) return v;
    return d.toLocaleDateString();
  }

  async function load(){
    list.innerHTML='<div class="empty">Loading tasks...</div>';

    const snap=await db.collection("tasks").get();

    tasks=snap.docs.map(d=>({id:d.id,...d.data()}))
      .sort((a,b)=>new Date(a.dueDate||0)-new Date(b.dueDate||0));

    render();
  }

  function render(){
    if(!tasks.length){
      list.innerHTML='<div class="empty">No tasks yet.</div>';
      return;
    }

    list.innerHTML="";

    tasks.forEach(t=>{
      const done=!!t.completed;
      const card=document.createElement("article");
      card.className="task";

      card.innerHTML=`
        <button class="check ${done?'done':''}" title="Mark complete">${done?'✓':''}</button>

        <div>
          <div class="title">${t.title||"Untitled Task"}</div>
          <div class="meta">Assigned to: ${t.assignedTo||"Unassigned"}</div>
          <div class="meta">Due: ${fmtDate(t.dueDate)}</div>
          ${t.completedAtText?`<div class="meta">Completed: ${t.completedAtText}</div>`:""}
          <span class="badge">${done?'Completed':'Pending'}</span>
        </div>

        <div class="actions">
          <button class="small edit">Edit</button>
          <button class="small delete">Delete</button>
        </div>
      `;

      card.querySelector(".check").onclick=async()=>{
        const now=new Date();
        await db.collection("tasks").doc(t.id).set({
          completed:!done,
          completedAt:!done ? now.toISOString() : "",
          completedAtText:!done ? now.toLocaleString() : "",
          updatedAt:now.toISOString()
        },{merge:true});
        await load();
      };

      card.querySelector(".edit").onclick=()=>{
        taskTitle.value=t.title||"";
        assignedTo.value=t.assignedTo||"";
        dueDate.value=t.dueDate||"";
        form.dataset.editing=t.id;
        taskTitle.focus();
      };

      card.querySelector(".delete").onclick=async()=>{
        if(confirm("Delete this task?")){
          await db.collection("tasks").doc(t.id).delete();
          await load();
        }
      };

      list.appendChild(card);
    });
  }

  form.onsubmit=async(e)=>{
    e.preventDefault();

    const now=new Date();

    const data={
      title:taskTitle.value.trim(),
      assignedTo:assignedTo.value.trim(),
      dueDate:dueDate.value,
      completed:false,
      updatedAt:now.toISOString(),
      updatedAtText:now.toLocaleString()
    };

    const editing=form.dataset.editing;

    if(editing){
      await db.collection("tasks").doc(editing).set(data,{merge:true});
      delete form.dataset.editing;
    }else{
      await db.collection("tasks").add({
        ...data,
        createdAt:now.toISOString(),
        createdAtText:now.toLocaleString()
      });
    }

    form.reset();
    await load();
  };

  await load();
});
</script>

  function addMenuItems(){
    const all = Array.from(document.querySelectorAll("a,button,div,li"));

    const kpiItem = all.find(el => (el.textContent || "").trim() === "Team KPIs");
    if(!kpiItem || !kpiItem.parentElement) return;

    const menu = kpiItem.parentElement;

    Array.from(menu.children).forEach(el => {
      const text = (el.textContent || "").trim();
      if(text === "Tasks" || text === "Team Meetings" || text === "Recycle Bin"){
        el.remove();
      }
    });

    const teamDocs = Array.from(menu.children).find(el =>
      (el.textContent || "").trim() === "Team Docs"
    );

    if(!teamDocs){
      const reports = Array.from(menu.children).find(el =>
        (el.textContent || "").trim() === "Reports"
      );

      if(reports){
        reports.textContent = "Team Docs";
        reports.href = "/teacher/team-docs.html";
      }
    }

    const operations = Array.from(menu.children).find(el =>
      (el.textContent || "").trim() === "Operations"
    );

    if(operations){
      menu.insertBefore(makeMenuLink("Tasks","/teacher/tasks.html",kpiItem), operations);
      menu.insertBefore(makeMenuLink("Team Meetings","/teacher/team-meetings.html",kpiItem), operations.nextSibling);
      menu.appendChild(makeMenuLink("Recycle Bin","/teacher/recycle-bin.html",kpiItem));
    }else{
      menu.appendChild(makeMenuLink("Tasks","/teacher/tasks.html",kpiItem));
      menu.appendChild(makeMenuLink("Team Meetings","/teacher/team-meetings.html",kpiItem));
      menu.appendChild(makeMenuLink("Recycle Bin","/teacher/recycle-bin.html",kpiItem));
    }
  }

  setTimeout(addMenuItems,100);
  setTimeout(addMenuItems,400);
  setTimeout(addMenuItems,1000);
  setTimeout(addMenuItems,1800);

<script id="team-meetings-menu-fix">
(function(){

  function injectTeamMeetings(){

    const menu =
      document.querySelector(".quick-menu") ||
      document.querySelector(".menu-dropdown") ||
      document.querySelector(".menu-panel") ||
      document.querySelector(".menu-links") ||
      document.querySelector(".dropdown-menu");

    if(!menu) return;

    if(document.getElementById("teamMeetingsMenuLink")) return;

    const recycleLink = Array.from(menu.querySelectorAll("a"))
      .find(a =>
        (a.textContent || "")
          .toLowerCase()
          .includes("recycle")
      );

    if(!recycleLink) return;

    const newLink = recycleLink.cloneNode(true);

    newLink.id = "teamMeetingsMenuLink";

    newLink.href = "/teacher/team-meetings.html";

    newLink.textContent = "Team Meetings";

    // REMOVE INLINE STYLES THAT CAUSED STATIC LOOK
    newLink.removeAttribute("style");

    // insert before recycle bin
    recycleLink.parentNode.insertBefore(newLink, recycleLink);

  }

  injectTeamMeetings();

  setInterval(injectTeamMeetings, 1500);

})();
</script>
