  function showDeletePopover(anchorRow, msg){
    // Remove any existing popover first
    document.querySelectorAll(".del-popover").forEach(p=>p.remove());

    return new Promise(resolve=>{
      const pop = document.createElement("div");
      pop.className = "del-popover";
      pop.innerHTML = `
        <span class="del-popover-msg">${msg}</span>
        <div class="del-popover-actions">
          <button class="del-popover-cancel" type="button">Cancel</button>
          <button class="del-popover-ok" type="button">Delete</button>
        </div>`;

      // Insert after the row that was clicked
      anchorRow.after(pop);
      pop.scrollIntoView({behavior:"smooth", block:"nearest"});

      const close = r=>{
        pop.remove();
        document.removeEventListener("click", outside);
        resolve(r);
      };

      pop.querySelector(".del-popover-ok").onclick     = e=>{ e.stopPropagation(); close(true); };
      pop.querySelector(".del-popover-cancel").onclick = e=>{ e.stopPropagation(); close(false); };

      // Click outside dismisses
      const outside = e=>{
        if(!pop.contains(e.target)) close(false);
      };
      setTimeout(()=>document.addEventListener("click", outside), 50);
    });
  }

  async function loadPlans(){

      const snap = await db.collection("activityPlans").orderBy("updatedAt","desc").get();

    render();
  }

let shareId = null;
let shareLearners = [];
let allLearnersCache = [];

async function loadLearnersOnce(){
  if(allLearnersCache.length) return;
  try{
    const snap = await db.collection("impactPasskeys").where("role","==","Learner").get();
    allLearnersCache = snap.docs.map(function(d){
      return { name: d.data().name || "Unnamed", level: d.data().level || "" };
    });
  }catch(e){ console.warn("learners load:", e); }
}

function renderShareChips(){
  const wrap = document.getElementById("shareSelected");
  if(!wrap) return;
  wrap.innerHTML = shareLearners.map(function(n){
    return '<span class="assigned-learner-chip">' + n + '<button type="button" data-rm="' + n + '">x</button></span>';
  }).join("");
}

async function openShareDialog(id, title){
  shareId = id;
  await loadLearnersOnce();
  const doc = await db.collection("activityPlans").doc(id).get();
  const d = doc.exists ? doc.data() : {};
  shareLearners = Array.isArray(d.assignedLearners) ? d.assignedLearners.slice() : [];

  document.getElementById("shareTitle").textContent = "Share: " + (title || "Activity");
  document.querySelectorAll("#shareLevels input").forEach(function(cb){
    cb.checked = Array.isArray(d.assignedLevels) && d.assignedLevels.indexOf(cb.value) >= 0;
    cb.closest(".assign-level-chip").classList.toggle("checked", cb.checked);
  });
  document.getElementById("shareSearch").value = "";
  document.getElementById("sharePicker").classList.remove("show");
  renderShareChips();
  document.getElementById("shareOverlay").classList.add("show");
}

document.getElementById("shareLevels").addEventListener("change", function(e){
  const chip = e.target.closest(".assign-level-chip");
  if(chip) chip.classList.toggle("checked", e.target.checked);
});

document.getElementById("shareSearch").addEventListener("input", function(){
  const q = this.value.trim().toLowerCase();
  const list = document.getElementById("sharePicker");
  if(!q){ list.classList.remove("show"); list.innerHTML=""; return; }
  const matches = allLearnersCache.filter(function(l){
    return l.name.toLowerCase().indexOf(q) >= 0 && shareLearners.indexOf(l.name) < 0;
  });
  if(!matches.length){ list.classList.remove("show"); list.innerHTML=""; return; }
  list.innerHTML = matches.map(function(l){
    return '<div class="learner-picker-row" data-pick="' + l.name + '">' + l.name + ' <span style="opacity:.5">(' + (l.level||"no level") + ')</span></div>';
  }).join("");
  list.classList.add("show");
});

document.getElementById("sharePicker").addEventListener("click", function(e){
  const row = e.target.closest("[data-pick]");
  if(!row) return;
  if(shareLearners.indexOf(row.dataset.pick) < 0) shareLearners.push(row.dataset.pick);
  renderShareChips();
  document.getElementById("shareSearch").value = "";
  this.classList.remove("show");
  this.innerHTML = "";
});

document.getElementById("shareSelected").addEventListener("click", function(e){
  const btn = e.target.closest("[data-rm]");
  if(!btn) return;
  shareLearners = shareLearners.filter(function(n){ return n !== btn.dataset.rm; });
  renderShareChips();
});

document.getElementById("shareCancel").addEventListener("click", function(){
  document.getElementById("shareOverlay").classList.remove("show");
});

  function patchFirestore(){
    if(!window.firebase || !firebase.firestore || firebase.__impactRecyclePatched) return;
    firebase.__impactRecyclePatched = true;

    const proto = firebase.firestore.DocumentReference.prototype;
    const originalDelete = proto.delete;

    proto.delete = async function(){
      let data = {};
      try{
        const snap = await this.get();
        if(snap.exists) data = snap.data() || {};
      }catch(e){}

      await window.ImpactRecycleBin.send({
        type: "firestore-document",
        title: data.title || data.name || data.filename || data.fileName || this.id,
        collection: this.parent && this.parent.id ? this.parent.id : "",
        documentId: this.id,
        data: data
      });

      return originalDelete.apply(this, arguments);
    };

  loadLearnersList();

  learnerSearchInput.addEventListener("input", function(){
    const q = learnerSearchInput.value.trim().toLowerCase();
    if(!q){ learnerPickerList.classList.remove("show"); learnerPickerList.innerHTML=""; return; }
    const matches = allLearners.filter(function(l){
      return l.name.toLowerCase().includes(q) && !selectedLearners.includes(l.name);
    });
    if(!matches.length){ learnerPickerList.classList.remove("show"); learnerPickerList.innerHTML=""; return; }
    learnerPickerList.innerHTML = matches.map(function(l){

function createBlock(type, data){
  const tmpl = TEMPLATES[type];
  if(!tmpl) return null;
  const block = document.createElement("div");
  block.className = "c-block";
  block.setAttribute("data-type", type);
  const actions = document.createElement("div");
  actions.className = "block-actions";
  actions.innerHTML = '<button class="act-btn danger" data-act="del" title="Delete block"><svg viewBox="0 0 24 24"><polyline points="3 6 5 6 21 6"/><path d="M19 6l-1 14a2 2 0 0 1-2 2H8a2 2 0 0 1-2-2L5 6"/><path d="M10 11v6M14 11v6"/></svg></button>';
  const inner = document.createElement("div");
  inner.className = "c-block-inner";
  inner.innerHTML = tmpl.render(data);
  block.appendChild(actions);
  block.appendChild(inner);
  bindBlock(block);
  return block;
}

function bindBlock(block){
  block.addEventListener("click", function(e){
    if(e.target.closest(".block-actions")) return;
    e.stopPropagation();
    selectBlock(block);
  });

  block.querySelector(".block-actions").addEventListener("click", function(e){
    e.stopPropagation();
    const btn = e.target.closest("[data-act]"); if(!btn) return;
    const act = btn.dataset.act;
    if(act==="del"){ pushUndo(); block.remove(); selectedBlock=null; showEmpty(); }
    if(act==="up"){ pushUndo(); const p=block.previousElementSibling; if(p&&p.classList.contains("c-block")) canvas.insertBefore(block,p); }
    if(act==="down"){ pushUndo(); const n=block.nextElementSibling; if(n&&n.classList.contains("c-block")) canvas.insertBefore(n,block); }
    if(act==="dupe"){ pushUndo(); const d=createBlock(block.dataset.type, extractData(block)); block.after(d); selectBlock(d); }
  });

  const audioZone = block.querySelector(".delf-audio-zone");
  const audioInput = block.querySelector(".delf-audio-input");
  const audioPlayer = block.querySelector(".delf-audio-player");
  const audioStatus = block.querySelector(".delf-audio-status");

  if(audioZone && audioInput){
    audioZone.addEventListener("click", function(e){ e.stopPropagation(); audioInput.click(); });
    audioInput.addEventListener("click", function(e){ e.stopPropagation(); });
    audioInput.addEventListener("change", async function(e){
      e.stopPropagation();
      const file = audioInput.files[0];
      if(!file) return;
      audioStatus.textContent = "Uploading...";
      try{
        const ext = file.name.split(".").pop() || "mp3";
        const path = "delfAudio/" + Date.now() + "_" + Math.random().toString(36).slice(2,8) + "." + ext;
        const snap = await storage.ref(path).put(file);
        const url = await snap.ref.getDownloadURL();
        block.dataset.audioSrc = url;
        audioPlayer.innerHTML = '<audio src="' + url + '" controls></audio>';
        audioPlayer.classList.add("show");
        audioZone.style.display = "none";
        audioStatus.textContent = "";
      }catch(err){
        audioStatus.textContent = "Upload failed: " + err.message;
      }
    });
  }

  const addQBtn = block.querySelector(".delf-add-question");
  if(addQBtn){
    addQBtn.addEventListener("click", function(e){
      e.stopPropagation();
      pushUndo();
      const wrap = block.querySelector(".delf-questions-wrap");
      const qid = "q"+Date.now()+"_"+Math.random().toString(36).slice(2,6);
      const div = document.createElement("div");
      div.innerHTML = questionRowHtml(qid, {question:"",options:["Option A","Option B","Option C","Option D"],correct:0});
      wrap.appendChild(div.firstElementChild);
    });
  }
  const qWrap = block.querySelector(".delf-questions-wrap");
  if(qWrap){
    qWrap.addEventListener("click", function(e){
      const rmBtn = e.target.closest(".delf-remove-question");
      if(rmBtn){
        e.stopPropagation();
        pushUndo();
        rmBtn.closest(".delf-question-row").remove();
      }
    });
  }
}

function extractData(block){
  const type = block.dataset.type;
  const data = { type: type };

  if(type==="sectionheader"){
    data.title = block.querySelector(".delf-section-title").innerHTML;
    data.points = block.querySelector(".delf-section-points").innerHTML;
  }
  if(type==="header"||type==="text"){
    data.content = block.querySelector("[contenteditable]").innerHTML;
  }
  if(type==="listening"||type==="reading"){
    const questions = Array.from(block.querySelectorAll(".delf-question-row")).map(function(row){
      const question = row.querySelector(".delf-question-text").innerHTML;
      const opts = Array.from(row.querySelectorAll(".delf-answer-opt"));
      const options = opts.map(function(o){ return o.querySelector("span").innerHTML; });
      let correct = 0;
      opts.forEach(function(o, i){ if(o.querySelector("input").checked) correct = i; });
      return { question: question, options: options, correct: correct };
    });
    data.questions = questions;
    if(type==="listening") data.audioSrc = block.dataset.audioSrc || "";
    if(type==="reading") data.passage = block.querySelector(".delf-passage-box").innerHTML;
  }
  if(type==="writing"){
    data.prompt = block.querySelector(".delf-prompt-box").innerHTML;
    data.wordCount = block.querySelector(".delf-wordcount").value;
    data.timeLimit = block.querySelector(".delf-timelimit").value;
  }
  if(type==="speaking"){
    data.prompt = block.querySelector(".delf-prompt-box").innerHTML;
    data.prepTime = block.querySelector(".delf-preptime").value;
    data.speakTime = block.querySelector(".delf-speaktime").value;
  }
  return data;
}

function serializeBlocks(){ return allBlocks().map(extractData); }

function rebuildFromBlocks(blocks){
  allBlocks().forEach(function(b){ b.remove(); });
  selectedBlock = null;
  if(!blocks || !blocks.length){ showEmpty(); return; }
  hideEmpty();
  blocks.forEach(function(data){
    const block = createBlock(data.type, data);
    if(!block) return;
    canvas.appendChild(block);
    if(data.type==="header"||data.type==="text"){
      const ce = block.querySelector("[contenteditable]");
      if(ce && data.content) ce.innerHTML = data.content;
    }
    if(data.type==="listening" && data.audioSrc){
      block.dataset.audioSrc = data.audioSrc;
    }
  });
}

async function save(){
  try{
    const blocks = serializeBlocks();
    const assignedLevels = Array.from(document.querySelectorAll("#assignLevelsGroup input:checked")).map(function(c){ return c.value; });
    const payload = {
      createdByTeacher: (sessionStorage.getItem("impactAccessName") || sessionStorage.getItem("impactPersonnelName") || ""),
      title: titleInput.value.trim() || "Untitled Mock Exam",
      track: trackSelect.value,
      level: levelSelect.value,
      blocks: blocks,
      version: 1,
      updatedAt: new Date().toISOString()
    };
    if(!currentId){ payload.status = "draft"; }
    if(currentId){
      await db.collection("delfMocks").doc(currentId).set(payload, {merge:true});
    } else {
      const ref = await db.collection("delfMocks").add(payload);
      currentId = ref.id;
      history.replaceState(null,"","/teacher/delf-mock-builder.html?edit="+currentId);
    }
    tick();
    return currentId;
  }catch(err){
    console.error("Save failed:", err);
    alert("Save failed: " + err.message);
    return null;
  }
}

let autoSaveTimer = null;
function scheduleAutoSave(){
  const hasTitle = titleInput.value.trim().length > 0;
  const hasBlocks = allBlocks().length > 0;
  if(!hasTitle || !hasBlocks) return;
  clearTimeout(autoSaveTimer);
  autoSaveTimer = setTimeout(async function(){ await save(); }, 1500);
}

trackSelect.addEventListener("change", scheduleAutoSave);
levelSelect.addEventListener("change", scheduleAutoSave);
new MutationObserver(function(){ scheduleAutoSave(); }).observe(canvas, {childList:true,subtree:false});

async function load(){
  if(!currentId) return;
  const doc = await db.collection("delfMocks").doc(currentId).get();
  if(!doc.exists) return;
  const d = doc.data();
  titleInput.value = d.title || "";
  if(d.track) trackSelect.value = d.track;
  if(d.level) levelSelect.value = d.level;
  if(Array.isArray(d.assignedLevels)){
    d.assignedLevels.forEach(function(lvl){
      const cb = document.querySelector('#assignLevelsGroup input[value="' + lvl + '"]');
      if(cb){ cb.checked = true; cb.closest(".assign-level-chip").classList.add("checked"); }
    });
  }
  if(Array.isArray(d.assignedLearners)){
    selectedLearners = d.assignedLearners.slice();
    renderSelectedLearnerChips();
  }
  currentStatus = d.status || "draft";
  renderStatus();
  if(Array.isArray(d.blocks)) rebuildFromBlocks(d.blocks);
}

    pill.className = "status-pill draft";
    note.textContent = "Not yet visible to learners.";
    pubBtn.disabled = false;

}
