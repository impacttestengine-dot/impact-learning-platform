document.addEventListener("DOMContentLoaded", function(){

  // Uses the same fuzzy name-matching as kpis.html so a short
  // passkey name (e.g. "Prosper") still correctly links to full
  // names used in class/task records (e.g. "Yormesah Prosper").
  window.addEventListener("load", async function(){

      function namesMatch(passkeyName, recordName){
        if(!passkeyName || !recordName) return false;
        const a = String(passkeyName).trim().toLowerCase();
        const b = String(recordName).trim().toLowerCase();
        if(!a || !b) return false;
        return a === b || b.includes(a) || a.includes(b);
      }

      function nearestMarkInfo(startedAt){
        const d = new Date(startedAt);
        const minutes = d.getMinutes();
        const lastMark = minutes < 30 ? 0 : 30;
        const nextMark = lastMark === 0 ? 30 : 60;
        const minutesUntilNextMark = nextMark - minutes;
        if(minutesUntilNextMark <= 5){
          const inferred = new Date(d);
          if(nextMark === 60){ inferred.setHours(inferred.getHours()+1, 0, 0, 0); }
          else{ inferred.setMinutes(30, 0, 0); }
          return inferred;
        }
        const inferred = new Date(d);
        inferred.setMinutes(lastMark, 0, 0);
        return inferred;
      }

      function computePunctuality(classes){
        const withStart = classes.filter(x=>x.startedAt);
        if(!withStart.length) return { pct: 0, count: 0 };
        let onTime = 0, counted = 0;
        withStart.forEach(x=>{
          const actual = new Date(x.startedAt);
          let agreedTime = null;
          if(x.time){
            const [schedH, schedM] = String(x.time).split(":").map(Number);
            if(!isNaN(schedH) && !isNaN(schedM)){
              const scheduled = new Date(actual);
              scheduled.setHours(schedH, schedM, 0, 0);
              if(Math.abs(actual - scheduled) / 60000 <= 35) agreedTime = scheduled;
            }
          }
          if(!agreedTime) agreedTime = nearestMarkInfo(x.startedAt);
          const diffMinutes = (actual - agreedTime) / 60000;
          counted++;
          if(diffMinutes <= 5) onTime++;
        });
        return { pct: counted ? Math.round((onTime/counted)*100) : 0, count: counted };
      }

      function tone(pct){ return pct>=80?"good":pct>=50?"warn":"bad"; }

      `;
    }catch(e){ console.warn("KPI snapshot failed:", e); }
  });

        documentId: this.id,
        data: data
      });

      return originalDelete.apply(this, arguments);
    };
  }

  function patchStorage(){
    if(!window.firebase || !firebase.storage || firebase.__impactStorageRecyclePatched) return;
    firebase.__impactStorageRecyclePatched = true;

    const proto = firebase.storage.Reference.prototype;
    const originalDelete = proto.delete;

    proto.delete = async function(){
      await window.ImpactRecycleBin.send({
        type: "storage-file",
        title: this.name || this.fullPath || "Deleted file",
        storagePath: this.fullPath || "",
        fileName: this.name || ""
      });

      return originalDelete.apply(this, arguments);
    };
  }

  function init(){
    patchFirestore();
    patchStorage();
    setTimeout(patchFirestore, 1000);
    setTimeout(patchStorage, 1000);
  }

  if(document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded", init);
  }else{
    init();
  }
})();
