document.addEventListener("DOMContentLoaded", function(){
  const db = firebase.firestore();

  function toast(msg, dur=2800){
    const t = document.getElementById("toast");
    t.textContent = msg;
    t.classList.add("show");
    setTimeout(()=>t.classList.remove("show"), dur);
  }

  function genId(){
    const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
    let id = "";
    for(let i=0;i<3;i++){
      for(let j=0;j<3;j++) id += chars[Math.floor(Math.random()*chars.length)];
      if(i<2) id += "-";
    }
    return id;
  }

  document.getElementById("createBtn").onclick = async function(){
    const name  = document.getElementById("createName").value.trim();
    const title = document.getElementById("meetingTitle").value.trim() || "Impact Live Class";
    const err   = document.getElementById("createErr");

    if(!name){ err.textContent = "Please enter your display name."; return; }
    err.textContent = "";

    const meetingId = genId();

    await db.collection("meetings").doc(meetingId).set({
      meetingId,
      title,
      createdAt:  new Date().toISOString(),
      startedAt:  new Date().toISOString(),
      endedAt:    "",
      status:     "active",
      hostName:   name
    });

    // window.open() must run synchronously inside the click handler --
    // mobile Safari blocks popups opened from inside a setTimeout, even a
    // short one, since it breaks the direct user-gesture chain it requires.
    window.open("/teacher/live-room.html?id=" + meetingId + "&name=" + encodeURIComponent(name) + "&host=true", "_blank");
    toast("Meeting created — opening in a new tab");
  };

  document.getElementById("joinBtn").onclick = async function(){
    const name = document.getElementById("joinName").value.trim();
    const id   = document.getElementById("joinId").value.trim().toUpperCase().replace(/\s/g,"");
    const err  = document.getElementById("joinErr");

    if(!name){ err.textContent = "Please enter your display name."; return; }
    if(!id)  { err.textContent = "Please enter a Meeting ID."; return; }
    err.textContent = "";

    try{
      const snap = await db.collection("meetings").doc(id).get();
      if(!snap.exists || snap.data().status !== "active"){
        err.textContent = "Meeting not found or has ended."; return;
      }
      window.open("/teacher/live-join.html?id=" + id + "&name=" + encodeURIComponent(name) + "&host=false", "_blank");
      toast("Opening meeting in a new tab");
    }catch(e){
      err.textContent = "Could not reach the meeting. Check your connection.";
    }
  };

  document.getElementById("joinId").addEventListener("keydown", function(e){
    if(e.key === "Enter") document.getElementById("joinBtn").click();
  });

  const params = new URLSearchParams(location.search);
  if(params.get("join")){
    document.getElementById("joinId").value = params.get("join");
    document.getElementById("joinName").focus();
  }
});
