// ══════════════════════════════════════════════════════════════
// LOCAL MEDIA — correct permission handling
// ══════════════════════════════════════════════════════════════
async function startLocalStream(){
  const AUDIO_CONSTRAINTS = { echoCancellation:true, noiseSuppression:true, autoGainControl:true };

  // Progressive fallback. "facingMode:user" and strict width/height/frameRate
  // throw OverconstrainedError on many laptop webcams, which previously dropped
  // straight to audio-only and left every video tile blank. Try loosest-last.
  const VIDEO_ATTEMPTS = [
    { facingMode:"user", width:{ideal:1280}, height:{ideal:720} },
    { width:{ideal:1280}, height:{ideal:720} },
    true
  ];

  let gotVideo = false;
  for(const videoConstraint of VIDEO_ATTEMPTS){
    try{
      localStream = await navigator.mediaDevices.getUserMedia({
        video: videoConstraint,
        audio: AUDIO_CONSTRAINTS
      });
      gotVideo = true;
      break;
    }catch(e){
      console.warn("getUserMedia attempt failed:", e.name, JSON.stringify(videoConstraint));
      if(e.name === "NotAllowedError") break; // permission denied - stop retrying
    }
  }

  try{
    if(gotVideo) throw { __skip:true };
    const err = { name: "OverconstrainedError" };
    try{
      localStream = await navigator.mediaDevices.getUserMedia({
        audio:{ echoCancellation:true, noiseSuppression:true }
      });
      camOn = false;
      toast("📷 Camera unavailable — joined with mic only.");
    }catch(err2){
      if(err2 && err2.__skip) throw err2;
      localStream = new MediaStream();
      camOn = false; micOn = false;
      if(err.name === "NotAllowedError" || err2.name === "NotAllowedError"){
        permBanner.classList.add("show");
        document.getElementById("permBannerText").innerHTML =
          'Camera/mic blocked. Click the 🔒 icon in your address bar → Allow → then <a href="javascript:location.reload()" style="color:#fde68a;text-decoration:underline">reload</a>.';
      }else{
        toast("No camera or microphone detected.");
      }
    }
  }catch(outer){
    if(!(outer && outer.__skip)) console.warn("media setup:", outer);
  }

  if(!localStream) localStream = new MediaStream();
  if(localStream.getVideoTracks().length === 0) camOn = false;

  localStream.getVideoTracks().forEach(t => t.enabled = camOn);
  localStream.getAudioTracks().forEach(t => t.enabled = micOn);

  ensureTile(MY_ID, MY_NAME, true, IS_HOST);
  setTileStream(MY_ID, localStream);
  setCamOff(MY_ID, !camOn || localStream.getVideoTracks().length===0);
  setMicIcon(MY_ID, micOn);

  camBtn.classList.toggle("off", !camOn);
  micBtn.classList.toggle("off", !micOn);
}

// ══════════════════════════════════════════════════════════════
// MESH WEBRTC — perfect negotiation, every peer ↔ every peer
// Fine for up to ~6 people; ideal for your 1-4 person classes.
// Signalling doc key = sorted pair of IDs so both sides agree.
// ══════════════════════════════════════════════════════════════
function sigDocId(otherId){
  return [MY_ID, otherId].sort().join("__");
}
function sigRef(otherId){
  return db.collection("meetings").doc(MEETING_ID).collection("signals").doc(sigDocId(otherId));
}
function iceOutRef(otherId){ return sigRef(otherId).collection("ice_"+MY_ID); }
function iceInRef(otherId){ return sigRef(otherId).collection("ice_"+otherId); }

async function createPeerConnection(otherId, otherName, otherIsHost){
  if(peerConns[otherId]) return peerConns[otherId];

  // Politeness: deterministic based on ID comparison so both sides agree.
  // The peer with the alphabetically LATER id is "polite" (yields on collision).
  const isPolite = MY_ID > otherId;
  politeMap[otherId] = isPolite;
  makingOffer[otherId] = false;

  const pc = new RTCPeerConnection(ICE_CONFIG);
  peerConns[otherId] = pc;
  peerMeta[otherId]  = { name: otherName, isHost: otherIsHost, camOn:true, micOn:true };

  // Add local tracks BEFORE any negotiation — critical for SDP to include media
  if(localStream){
    localStream.getTracks().forEach(track => pc.addTrack(track, localStream));
  }

  ensureTile(otherId, otherName, false, otherIsHost);

  pc.ontrack = (event) => {
    const stream = event.streams[0];
    if(!stream) return;
    setTileStream(otherId, stream);
    setCamOff(otherId, stream.getVideoTracks().length === 0 || !stream.getVideoTracks()[0].enabled);
  };

  pc.onicecandidate = (event) => {
    if(!event.candidate) return;
    iceOutRef(otherId).add({ candidate: event.candidate.toJSON(), ts: Date.now() }).catch(()=>{});
  };

  pc.onnegotiationneeded = async () => {
    try{
      makingOffer[otherId] = true;
      await pc.setLocalDescription();
      await sigRef(otherId).set({
        [`sdp_${MY_ID}`]: { type: pc.localDescription.type, sdp: pc.localDescription.sdp }
      }, { merge:true });
    }catch(e){
      console.error("onnegotiationneeded error:", e);
    }finally{
      makingOffer[otherId] = false;
    }
  };

  pc.onconnectionstatechange = () => {
    const s = pc.connectionState;
    if(s === "connected"){
      const total = Object.keys(peerConns).length + 1;
      setStatus(`Live · ${total} in class`);
    }
    if(s === "failed"){
      toast(`⚠️ Connection to ${otherName} failed. They may need to rejoin.`);
    }
  };

  // Listen for remote SDP (offer or answer — perfect negotiation treats both uniformly)
  const unsubSdp = sigRef(otherId).onSnapshot(async (snap) => {
    const data = snap.data();
    if(!data) return;
    const remoteSdp = data[`sdp_${otherId}`];
    if(!remoteSdp || !remoteSdp.sdp) return;

    // CRITICAL: this snapshot also fires when WE write our own SDP into the
    // same doc. Without this guard we re-apply the same remote SDP forever,
    // which throws "wrong state: stable" and blocks the connection.
    if(lastAppliedSdp[otherId] === remoteSdp.sdp) return;

    try{
      const offerCollision = remoteSdp.type === "offer" &&
        (makingOffer[otherId] || pc.signalingState !== "stable");

      const ignoreOffer = !isPolite && offerCollision;
      if(ignoreOffer) return;

      await pc.setRemoteDescription(new RTCSessionDescription(remoteSdp));
      lastAppliedSdp[otherId] = remoteSdp.sdp;

      if(remoteSdp.type === "offer"){
        await pc.setLocalDescription();
        await sigRef(otherId).set({
          [`sdp_${MY_ID}`]: { type: pc.localDescription.type, sdp: pc.localDescription.sdp }
        }, { merge:true });
      }
    }catch(e){
      console.error("SDP handling error:", e);
    }
  });
  unsubs.push(unsubSdp);

  // Listen for inbound ICE candidates
  const unsubIce = iceInRef(otherId).orderBy("ts").onSnapshot((snap) => {
    snap.docChanges().forEach(async (change) => {
      if(change.type === "added"){
        try{ await pc.addIceCandidate(new RTCIceCandidate(change.doc.data().candidate)); }
        catch(e){ /* benign during renegotiation */ }
      }
    });
  });
  unsubs.push(unsubIce);

  renderParticipantList();
  return pc;
}

function closePeerConnection(otherId){
  const pc = peerConns[otherId];
  if(pc){ try{ pc.close(); }catch(e){} }
  delete peerConns[otherId];
  delete peerMeta[otherId];
  removeTile(otherId);
  renderParticipantList();
}

// PARTICIPANT DISCOVERY — mesh: everyone connects to everyone
// ══════════════════════════════════════════════════════════════
function watchParticipants(){
  const unsub = db.collection("meetings").doc(MEETING_ID)
    .collection("participants")
    .onSnapshot((snap) => {
      snap.docChanges().forEach(async (change) => {
        const data = change.doc.data();
        const rid  = data.participantId;
        if(rid === MY_ID) return;

        if(change.type === "added"){
          toast(data.displayName + " joined the class");
          await createPeerConnection(rid, data.displayName, data.isHost);
        }

        if(change.type === "modified"){
          if(peerMeta[rid]){
            peerMeta[rid].camOn = data.cameraEnabled;
            peerMeta[rid].micOn = data.micEnabled;
            setCamOff(rid, !data.cameraEnabled);
            setMicIcon(rid, data.micEnabled);
            renderParticipantList();
          }
        }

        if(change.type === "removed"){
          toast(data.displayName + " left");
          closePeerConnection(rid);
          if(data.isHost){
            cleanup();
            location.href = "/teacher/live-ended.html?reason=ended";
          }
        }
      });
    });
  unsubs.push(unsub);
}

async function registerSelf(){
  await db.collection("meetings").doc(MEETING_ID)
    .collection("participants").doc(MY_ID).set({
      participantId: MY_ID, displayName: MY_NAME, isHost: IS_HOST,
      cameraEnabled: camOn, micEnabled: micOn, joinedAt: new Date().toISOString(),
      forceMuted: false, forceCameraOff: false, kicked: false
    });
  window.addEventListener("pagehide", cleanup);
  window.addEventListener("beforeunload", cleanup);

  // Watch my own doc for host moderation actions targeting me
  const unsubSelf = db.collection("meetings").doc(MEETING_ID)
    .collection("participants").doc(MY_ID)
    .onSnapshot((snap) => {
      const data = snap.data();
      if(!data) return;

      // Host force-muted me
      if(data.forceMuted && micOn){
        micOn = false;
        if(localStream) localStream.getAudioTracks().forEach(t => t.enabled = false);
        micBtn.classList.add("off");
        setMicIcon(MY_ID, false);
        toast("🔇 The host muted your microphone");
        db.collection("meetings").doc(MEETING_ID).collection("participants").doc(MY_ID)
          .update({ forceMuted: false }).catch(()=>{});
      }

      // Host turned off my camera
      if(data.forceCameraOff && camOn){
        camOn = false;
        if(localStream) localStream.getVideoTracks().forEach(t => t.enabled = false);
        camBtn.classList.add("off");
        setCamOff(MY_ID, true);
        toast("📷 The host turned off your camera");
        db.collection("meetings").doc(MEETING_ID).collection("participants").doc(MY_ID)
          .update({ forceCameraOff: false }).catch(()=>{});
      }

      // Host removed me from the meeting
      if(data.kicked){
        toast("You have been removed from the meeting by the host.");
        setTimeout(()=>{
          cleanup();
          location.href = "/teacher/live-ended.html?reason=removed";
        }, 1200);
      }
    });
  unsubs.push(unsubSelf);
}

  db.collection("meetings").doc(MEETING_ID).collection("participants").doc(MY_ID)
    .update({ cameraEnabled: camOn, micEnabled: micOn }).catch(()=>{});
  renderParticipantList();
}

// Detect if screen sharing is supported (desktop only — iOS/Android
// browsers do not implement getDisplayMedia at all)
const SCREEN_SHARE_SUPPORTED = !!(navigator.mediaDevices && navigator.mediaDevices.getDisplayMedia);

if(!SCREEN_SHARE_SUPPORTED){
  screenBtn.style.opacity = "0.35";
  screenBtn.title = "Screen sharing is only available on desktop browsers";
}

screenBtn.onclick = async function(){
  if(!SCREEN_SHARE_SUPPORTED){
    toast("📵 Screen sharing isn't supported on this device. Try from a desktop browser.");
    return;
  }
  if(screenOn){ stopScreenShare(); return; }
  try{
    screenStream = await navigator.mediaDevices.getDisplayMedia({
      video:{ cursor:"always", frameRate:15 },
      audio:true
    });
    screenOn = true;
    screenBtn.classList.add("active");
    toast("Screen sharing started");

    const screenTrack = screenStream.getVideoTracks()[0];

    // Replace outgoing video track on every peer connection
    await Promise.all(Object.values(peerConns).map(async pc => {
      const sender = pc.getSenders().find(s => s.track && s.track.kind === "video");
      if(sender) await sender.replaceTrack(screenTrack);
    }));

    // Show the screen share in the LOCAL tile
    setTileStream(MY_ID, screenStream);
    document.getElementById("tile-"+MY_ID).classList.remove("mirrored");

    screenTrack.onended = () => stopScreenShare();
  }catch(e){
    if(e.name !== "NotAllowedError") toast("Screen sharing failed: " + e.message);
  }
};

async function stopScreenShare(){
  if(!screenOn) return;
  screenOn = false;
  screenBtn.classList.remove("active");
  if(screenStream){ screenStream.getTracks().forEach(t=>t.stop()); screenStream=null; }

  if(localStream){
    const camTrack = localStream.getVideoTracks()[0];
    if(camTrack){
      await Promise.all(Object.values(peerConns).map(async pc => {
        const sender = pc.getSenders().find(s => s.track && s.track.kind === "video");
        if(sender) await sender.replaceTrack(camTrack);
      }));
    }
    setTileStream(MY_ID, localStream);
    document.getElementById("tile-"+MY_ID).classList.add("mirrored");
  }
  toast("Screen sharing stopped");
}

// ── PRESENT LESSON (host only) ──────────────────────────────
(function(){
  const lessonBtn = document.getElementById("lessonBtn");
  const lessonPanel = document.getElementById("lessonPanel");
  const lessonPanelClose = document.getElementById("lessonPanelClose");
  const lessonPicker = document.getElementById("lessonPicker");
  const lessonBlockList = document.getElementById("lessonBlockList");
  const presentedOverlay = document.getElementById("presentedOverlay");
  const presentedCard = document.getElementById("presentedCard");
  const presentedStopBtn = document.getElementById("presentedStopBtn");

  if(!IS_HOST) return; // only the host can present

  lessonBtn.style.display = "flex";

  lessonBtn.onclick = () => lessonPanel.classList.toggle("show");
  lessonPanelClose.onclick = () => lessonPanel.classList.remove("show");

  let currentLessonBlocks = [];
  let currentlyShowingIndex = -1;

  async function loadTeacherLessons(){
    try{
      const snap = await db.collection("lessonPlans")
        .where("createdByTeacher","==",MY_NAME)
        .get();
      const plans = snap.docs.map(d => ({ id: d.id, ...d.data() }));
      lessonPicker.innerHTML = '<option value="">Select a lesson plan...</option>' +
        plans.map(p => `<option value="${p.id}">${p.title || "Untitled Lesson"}</option>`).join("");
      lessonPicker._plans = plans;
    }catch(e){
      console.warn("Could not load lesson plans:", e);
    }
  }

  function renderBlockPreview(b){
    if(b.type === "mediafile" && b.src){
      return b.kind === "video"
        ? `<video src="${b.src}" muted style="width:100%;height:70px;object-fit:cover;border-radius:8px"></video>`
        : `<img src="${b.src}" style="width:100%;height:70px;object-fit:cover;border-radius:8px">`;
    }
    if(b.type === "header") return (b.content || "Header").replace(/<[^>]+>/g,"");
    if(b.type === "subheader") return (b.content || "Subheader").replace(/<[^>]+>/g,"");
    if(b.type === "text") return (b.content || "Text").replace(/<[^>]+>/g,"").slice(0,80);
    if(b.type === "mc" || b.type === "activity_mc") return (b.question || "Multiple choice question").replace(/<[^>]+>/g,"");
    if(b.type === "short" || b.type === "long") return (b.question || "Question").replace(/<[^>]+>/g,"");
    return "Block";
  }

  function renderBlockList(){
    lessonBlockList.innerHTML = currentLessonBlocks.map((b, i) => `
      <div class="lesson-block-item ${i===currentlyShowingIndex?'showing':''}">
        <div class="lesson-block-type">${(b.type||"block")}</div>
        <div class="lesson-block-preview">${renderBlockPreview(b)}</div>
        <button class="lesson-block-showbtn ${i===currentlyShowingIndex?'active':''}" data-idx="${i}">
          ${i===currentlyShowingIndex ? "Showing to class" : "Show to Class"}
        </button>
      </div>
    `).join("");

    lessonBlockList.querySelectorAll(".lesson-block-showbtn").forEach(btn => {
      btn.onclick = () => {
        const idx = Number(btn.dataset.idx);
        if(currentlyShowingIndex === idx){
          stopPresenting();
        }else{
          presentBlock(idx);
        }
      };
    });
  }

  lessonPicker.onchange = async () => {
    const id = lessonPicker.value;
    if(!id){ currentLessonBlocks = []; lessonBlockList.innerHTML = ""; return; }
    const plan = (lessonPicker._plans || []).find(p => p.id === id);
    currentLessonBlocks = (plan && Array.isArray(plan.blocks)) ? plan.blocks : [];
    currentlyShowingIndex = -1;
    renderBlockList();
  };

  async function presentBlock(idx){
    currentlyShowingIndex = idx;
    renderBlockList();
    await db.collection("meetings").doc(MEETING_ID).update({
      presentedBlock: currentLessonBlocks[idx],
      presentedAt: new Date().toISOString()
    }).catch(()=>{});
  }

  async function stopPresenting(){
    currentlyShowingIndex = -1;
    renderBlockList();
    await db.collection("meetings").doc(MEETING_ID).update({
      presentedBlock: null
    }).catch(()=>{});
  }

  presentedStopBtn.onclick = stopPresenting;

  loadTeacherLessons();
})();

// ── LISTEN FOR PRESENTED CONTENT (all participants, including host) ──
(function(){
  const presentedOverlay = document.getElementById("presentedOverlay");
  const presentedCard = document.getElementById("presentedCard");
  const presentedStopBtn = document.getElementById("presentedStopBtn");

  function renderPresentedBlock(b){
    if(!b) return "";
    if(b.type === "mediafile" && b.src){
      return b.kind === "video"
        ? `<video src="${b.src}" controls autoplay style="width:100%"></video>`
        : `<img src="${b.src}">`;
    }
    if(b.type === "header") return `<div class="p-header">${b.content||""}</div>`;
    if(b.type === "subheader") return `<div class="p-subheader">${b.content||""}</div>`;
    if(b.type === "text") return `<div class="p-text">${b.content||""}</div>`;
    if(b.type === "mc" || b.type === "activity_mc"){
      const opts = (b.options||[]).map(o => `<div class="p-opt">${o}</div>`).join("");
      return `<div class="p-question">${b.question||""}</div>${opts}`;
    }
    if(b.type === "short" || b.type === "long"){
      return `<div class="p-question">${b.question||""}</div>`;
    }
    return "";
  }

  db.collection("meetings").doc(MEETING_ID).onSnapshot(function(snap){
    const data = snap.data();
    if(!data) return;
    const block = data.presentedBlock;

    if(block){
      presentedCard.innerHTML = renderPresentedBlock(block);
      presentedOverlay.classList.add("active");
      document.body.classList.add("presenting");
      if(IS_HOST) presentedStopBtn.style.display = "inline-block";
    }else{

function startRecording(){
  recordChunks = [];
  const mime = MediaRecorder.isTypeSupported("video/webm;codecs=vp9,opus")
    ? "video/webm;codecs=vp9,opus"
    : MediaRecorder.isTypeSupported("video/webm") ? "video/webm" : "video/mp4";

  const canvas = document.createElement("canvas");
  canvas.width = 1280; canvas.height = 720;
  const ctx = canvas.getContext("2d");

  // Draw each tile preserving its native aspect ratio (contain-fit),
  // centred within its grid cell, instead of stretching to fill it.
  // This fixes the distorted/cropped recording seen on portrait mobile video.
  function drawTileContain(video, cellX, cellY, cellW, cellH){
    const vw = video.videoWidth  || 16;
    const vh = video.videoHeight || 9;
    if(!vw || !vh) return;

    const videoAspect = vw / vh;
    const cellAspect   = cellW / cellH;

    let drawW, drawH;
    if(videoAspect > cellAspect){
      // Video is relatively wider than the cell — fit to width
      drawW = cellW;
      drawH = cellW / videoAspect;
    }else{
      // Video is relatively taller than the cell — fit to height
      drawH = cellH;
      drawW = cellH * videoAspect;
    }

    const offsetX = cellX + (cellW - drawW) / 2;
    const offsetY = cellY + (cellH - drawH) / 2;

    ctx.drawImage(video, offsetX, offsetY, drawW, drawH);
  }

  const draw = setInterval(()=>{
    const tiles = Array.from(videoGrid.querySelectorAll("video")).filter(v=>v.srcObject && v.readyState>=2);
    const n = tiles.length || 1;
    const cols = n===1?1:2, rows = Math.ceil(n/cols);
    const tw = canvas.width/cols, th = canvas.height/rows;
    ctx.fillStyle="#0a0c14"; ctx.fillRect(0,0,canvas.width,canvas.height);
    tiles.forEach((v,i)=>{
      const col=i%cols, row=Math.floor(i/cols);
      try{ drawTileContain(v, col*tw+2, row*th+2, tw-4, th-4); }catch(e){}
    });
  }, 1000/24);

  const cs = canvas.captureStream(24);

  try{
    const ac = new AudioContext();
    const dest = ac.createMediaStreamDestination();
    if(localStream && localStream.getAudioTracks().length){
      ac.createMediaStreamSource(new MediaStream([localStream.getAudioTracks()[0]])).connect(dest);
    }
    Object.values(peerConns).forEach(pc=>{
      pc.getReceivers().filter(r=>r.track&&r.track.kind==="audio").forEach(r=>{
        try{ ac.createMediaStreamSource(new MediaStream([r.track])).connect(dest); }catch(e){}
      });
    });
    if(dest.stream.getAudioTracks().length) cs.addTrack(dest.stream.getAudioTracks()[0]);
  }catch(e){}

  try{ mediaRecorder = new MediaRecorder(cs,{mimeType:mime,videoBitsPerSecond:3000000}); }
  catch(e){ mediaRecorder = new MediaRecorder(cs); }

  mediaRecorder.ondataavailable = e=>{ if(e.data.size>0) recordChunks.push(e.data); };
  mediaRecorder.onstop = async ()=>{ clearInterval(draw); await uploadRecording(mime); };
  mediaRecorder.start(1000);
  recording = true;
  recBtn.classList.add("recording");
  toast("🔴 Recording started");
}

function stopRecording(){
  if(!mediaRecorder||!recording) return;
  recordingBusy = true;
  mediaRecorder.stop();
  recording = false;
  recBtn.classList.remove("recording");
  recBtn.disabled = true;
  toast("Uploading recording...", 5000);
}

async function uploadRecording(mime){
  // Snapshot the chunks locally BEFORE anything else can touch recordChunks,
  // in case a new recording starts while this upload is still in flight.
  const chunksToUpload = recordChunks.slice();
  recordChunks = [];

  try{
    if(!chunksToUpload.length){
      throw new Error("No video data was captured — recording was too short or a tab was backgrounded.");
    }

    const ext = mime.includes("mp4") ? "mp4" : "webm";
    // Collision-proof unique ID: timestamp + random suffix, guarantees
    // no two recordings in the same meeting can ever share a Storage path,
    // even if stopped/started within the same millisecond.
    const uniqueId = `${Date.now()}_${Math.random().toString(36).slice(2,8)}`;
    const path = `recordings/${MEETING_ID}/rec_${uniqueId}.${ext}`;

    const blob = new Blob(chunksToUpload, { type: mime });
    const snap = await storage.ref(path).put(blob);
    const url  = await snap.ref.getDownloadURL();
    const dur  = Math.round((Date.now()-startTime)/1000);

    if(pendingRecDestination === "placement"){
      await db.collection("placementRecordings").add({
        learnerName: pendingRecLearner, level: pendingRecLevel,
        fileName: meetingTitle || "Live Classroom Recording",
        url: url, storagePath: path,
        uploadedBy: MY_NAME, uploadedAt: new Date().toISOString(),
        sourceMeetingId: MEETING_ID
      });
    }else{
      await db.collection("recordings").add({
        meetingId:MEETING_ID, meetingTitle, storagePath:path, downloadURL:url,
        duration:dur, mimeType:mime, createdAt:new Date().toISOString(), recordedBy:MY_NAME
      });
    }
    await db.collection("meetings").doc(MEETING_ID).update({ hasRecording:true }).catch(()=>{});
    toast("✅ Recording saved!", 4000);
  }catch(e){
    console.error("Recording upload failed:", e);
    toast("⚠️ Recording failed to save: " + e.message, 6000);
  }finally{
    recBtn.disabled = false;
    recordingBusy = false;
  }
}
