document.addEventListener("DOMContentLoaded", async function(){
  const db = firebase.firestore();
  const wrap = document.getElementById("foldersWrap");
  const folderCount = document.getElementById("folderCount");
  let allPlans = [];
  let openFolder = null;

  function showDeletePopover(anchorRow, msg){
    // Remove any existing popover first
    document.querySelectorAll(".del-popover").forEach(p=>p.remove());

    return new Promise(resolve=>{
      const pop = document.createElement("div");
      pop.className = "del-popover";
      pop.innerHTML = `
        <span class="del-popover-msg">${msg}</span>
        <div class="del-popover-actions">
          <button class="del-popover-cancel" type="button">Cancel</button>
          <button class="del-popover-ok" type="button">Delete</button>
        </div>`;

      // Insert after the row that was clicked
      anchorRow.after(pop);
      pop.scrollIntoView({behavior:"smooth", block:"nearest"});

      const close = r=>{
        pop.remove();
        document.removeEventListener("click", outside);
        resolve(r);
      };

      pop.querySelector(".del-popover-ok").onclick     = e=>{ e.stopPropagation(); close(true); };
      pop.querySelector(".del-popover-cancel").onclick = e=>{ e.stopPropagation(); close(false); };

      // Click outside dismisses
      const outside = e=>{
        if(!pop.contains(e.target)) close(false);
      };
      setTimeout(()=>document.addEventListener("click", outside), 50);
    });
  }

  async function load(){
    try{
      const snap = await db.collection("lessonPlans").orderBy("updatedAt","desc").get();
      allPlans = snap.docs.map(d=>({id:d.id,...d.data()}));
    }catch(e){
      const snap = await db.collection("lessonPlans").get();
      allPlans = snap.docs.map(d=>({id:d.id,...d.data()}));
    }
    render();
  }

  function render(){
    // Group by level — each unique level = one folder
    const map = {};
    allPlans.forEach(p=>{
      const lvl = (p.level||"Uncategorised").trim();
      if(!map[lvl]) map[lvl]=[];
      map[lvl].push(p);
    });
    const levels = Object.keys(map).sort();

    folderCount.textContent = levels.length
      ? levels.length + " folder" + (levels.length===1?"":"s")
      : "";

    if(!levels.length){
      wrap.innerHTML = `<div class="empty-state">
        <h3>No folders yet</h3>
        <p>Create a lesson plan in the builder and select a level.<br>The folder will appear here automatically.</p>
      </div>`;
      return;
    }

    wrap.innerHTML = `<div class="folders-grid" id="grid"></div>`;
    const grid = document.getElementById("grid");

    levels.forEach(level=>{
      const plans = map[level];
      const card = document.createElement("div");
      card.className = "folder-card" + (openFolder===level?" open":"");
      card.dataset.level = level;

      const rows = plans.map(p=>`
        <div class="plan-row">
          <span class="plan-name">${p.title||p.name||"Untitled Lesson"}</span>
          <button class="plan-btn view-btn" data-id="${p.id}" title="Preview">👁</button>
          <button class="plan-btn edit-btn" data-id="${p.id}" title="Edit">✏️</button>
          <button class="plan-btn del del-btn" data-id="${p.id}" data-title="${(p.title||p.name||"Untitled Lesson").replace(/"/g,"&quot;")}" title="Delete">✕</button>
        </div>`).join("");

      card.innerHTML = `
        <div class="folder-head">
          <span class="folder-icon">${openFolder===level?"📂":"📁"}</span>
          <div class="folder-meta">
            <div class="folder-name">${level}</div>
            <div class="folder-sub">${plans.length} lesson plan${plans.length===1?"":"s"}</div>
          </div>
          <span class="folder-chevron">›</span>
        </div>
        <div class="folder-body">${rows}</div>`;

      grid.appendChild(card);
    });

    bindEvents();
  }

  function bindEvents(){
    const grid = document.getElementById("grid");
    if(!grid) return;

    grid.addEventListener("click", async function(e){

      // Delete
      const delBtn = e.target.closest(".del-btn");
      if(delBtn){
        e.stopPropagation();
        const anchorRow = delBtn.closest(".plan-row");
        const ok = await showDeletePopover(anchorRow, `Delete &ldquo;${delBtn.dataset.title}&rdquo;?`);
        if(!ok) return;
        await db.collection("lessonPlans").doc(delBtn.dataset.id).delete();
        allPlans = allPlans.filter(p=>p.id!==delBtn.dataset.id);
        const cardLevel = delBtn.closest(".folder-card").dataset.level;
        if(!allPlans.find(p=>(p.level||"Uncategorised")===cardLevel)) openFolder=null;
        render();
        return;
      }

      // Preview
      const viewBtn = e.target.closest(".view-btn");
      if(viewBtn){
        e.stopPropagation();
        location.href = "/teacher/lesson-view.html?id=" + encodeURIComponent(viewBtn.dataset.id);
        return;
      }

      // Edit
      const editBtn = e.target.closest(".edit-btn");
      if(editBtn){
        e.stopPropagation();
        location.href = "/teacher/lesson-builder.html?edit=" + encodeURIComponent(editBtn.dataset.id);
        return;
      }

      // Toggle folder
      const card = e.target.closest(".folder-card");
      if(card){
        const level = card.dataset.level;
        if(openFolder===level){
          openFolder = null;
          card.classList.remove("open");
          card.querySelector(".folder-icon").textContent = "📁";
        } else {
          document.querySelectorAll(".folder-card.open").forEach(c=>{
            c.classList.remove("open");
            c.querySelector(".folder-icon").textContent = "📁";
          });
          openFolder = level;
          card.classList.add("open");
          card.querySelector(".folder-icon").textContent = "📂";
        }
      }
    });

    // Click outside closes folders
    document.addEventListener("click", function(e){
      if(!e.target.closest(".folder-card") && !e.target.closest(".confirm-card")){
        document.querySelectorAll(".folder-card.open").forEach(c=>{
          c.classList.remove("open");
          const fi = c.querySelector(".folder-icon");
          if(fi) fi.textContent = "📁";
        });
        openFolder = null;
      }
    });
  }

  await load();
});

(function(){
  window.ImpactRecycleBin = {
    async send(item){
      try{
        if(!window.firebase || !firebase.firestore) return;
        const db = firebase.firestore();
        await db.collection("recycleBin").add({
          ...item,
          deletedAt: new Date().toISOString(),
          deletedAtText: new Date().toLocaleString(),
          sourceUrl: location.pathname,
          restored: false
        });
      }catch(e){
        console.warn("Recycle bin log failed", e);
      }
    }
  };

  function patchFirestore(){
    if(!window.firebase || !firebase.firestore || firebase.__impactRecyclePatched) return;
    firebase.__impactRecyclePatched = true;

    const proto = firebase.firestore.DocumentReference.prototype;
    const originalDelete = proto.delete;

    proto.delete = async function(){
      let data = {};
      try{
        const snap = await this.get();
        if(snap.exists) data = snap.data() || {};
      }catch(e){}

      await window.ImpactRecycleBin.send({
        type: "firestore-document",
        title: data.title || data.name || data.filename || data.fileName || this.id,
        collection: this.parent && this.parent.id ? this.parent.id : "",
        documentId: this.id,
        data: data
      });

      return originalDelete.apply(this, arguments);
    };
  }

  function patchStorage(){
    if(!window.firebase || !firebase.storage || firebase.__impactStorageRecyclePatched) return;
    firebase.__impactStorageRecyclePatched = true;

    const proto = firebase.storage.Reference.prototype;
    const originalDelete = proto.delete;

    proto.delete = async function(){
      await window.ImpactRecycleBin.send({
        type: "storage-file",
        title: this.name || this.fullPath || "Deleted file",
        storagePath: this.fullPath || "",
        fileName: this.name || ""
      });

      return originalDelete.apply(this, arguments);
    };
  }

  function init(){
    patchFirestore();
    patchStorage();
    setTimeout(patchFirestore, 1000);
    setTimeout(patchStorage, 1000);
  }

  if(document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded", init);
  }else{
    init();
  }
})();

document.addEventListener("DOMContentLoaded", async function(){
  const db = firebase.firestore();
  const name  = sessionStorage.getItem("impactAccessName") || sessionStorage.getItem("impactLearnerName") || "";
  const level = sessionStorage.getItem("impactAccessLevel") || "";

  const pendingWrap = document.getElementById("pendingWrap");
  const foldersWrap = document.getElementById("foldersWrap");

  if(!name){
    pendingWrap.innerHTML = '<div class="empty-state"><h3>Please sign in</h3></div>';
    foldersWrap.innerHTML = "";
    return;
  }

  try{
    const [byLevel, byName, completedSnap] = await Promise.all([
      level ? db.collection("lessonPlans").where("assignedLevels","array-contains",level).get() : Promise.resolve({docs:[]}),
      db.collection("lessonPlans").where("assignedLearners","array-contains",name).get(),
      db.collection("learnerCompletions").where("learnerName","==",name).where("itemType","==","lesson").get()
    ]);

    const completedIds = new Set();
    completedSnap.forEach(d=>completedIds.add(d.data().itemId));

    const itemMap = new Map();
    byLevel.docs.forEach(d=>itemMap.set(d.id, {id:d.id, ...d.data()}));
    byName.docs.forEach(d=>itemMap.set(d.id, {id:d.id, ...d.data()}));

    const allItems = Array.from(itemMap.values());
    const pending = allItems.filter(it=>!completedIds.has(it.id));
    const completed = allItems.filter(it=>completedIds.has(it.id));

    // ── PENDING ──
    if(!pending.length){
      pendingWrap.innerHTML = '<div class="empty-pending">✨ Nothing pending right now — you're all caught up!</div>';
    }else{
      pendingWrap.innerHTML = pending.map(it=>`
        <a class="pending-card" href="/teacher/=+it.id+">
          <div class="pending-icon">📚</div>
          <div class="pending-info">
            <div class="pending-title">+(it.title||"Untitled")+</div>
            <div class="pending-meta">+(it.level?"Level "+it.level:"")+</div>
          </div>
          <div class="pending-arrow">→</div>
        </a>
      `).join("");
    }

    // ── COMPLETED — grouped into folders by level ──
    if(!completed.length){
      foldersWrap.innerHTML = '<div class="empty-state"><h3>No completed items yet</h3></div>';
    }else{
      const byLevelMap = {};
      completed.forEach(it=>{
        const lvl = it.level || "Other";
        if(!byLevelMap[lvl]) byLevelMap[lvl] = [];
        byLevelMap[lvl].push(it);
      });

      foldersWrap.innerHTML = '<div class="folders-grid" id="grid"></div>';
      const grid = document.getElementById("grid");
      let openFolder = null;

      Object.keys(byLevelMap).sort().forEach(lvl=>{
        const items = byLevelMap[lvl];
        const card = document.createElement("div");
        card.className = "folder-card";
        card.dataset.level = lvl;

        const rows = items.map(it=>`
          <a class="completed-row" href="/teacher/=+it.id+">
            <span class="completed-check">✓</span>+(it.title||"Untitled")+
          </a>
        `).join("");

        card.innerHTML = `
          <div class="folder-head">
            <span class="folder-icon">📁</span>
            <div class="folder-meta">
              <div class="folder-name">+lvl+</div>
              <div class="folder-sub">+items.length+ completed</div>
            </div>
            <span class="folder-chevron">›</span>
          </div>
          <div class="folder-body">+rows+</div>
        `;

        card.querySelector(".folder-head").addEventListener("click", function(){
          const wasOpen = card.classList.contains("open");
          document.querySelectorAll(".folder-card.open").forEach(c=>c.classList.remove("open"));
          if(!wasOpen) card.classList.add("open");
        });

        grid.appendChild(card);
      });
    }
  }catch(e){
    console.error(e);
    pendingWrap.innerHTML = '<div class="empty-state"><h3>Could not load items</h3></div>';
    foldersWrap.innerHTML = "";
  }
});
</script>
